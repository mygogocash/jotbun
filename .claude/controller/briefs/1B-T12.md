## Interfaces (as `real/` has them)

```ts
// test/support/files-session.ts
export type FilesUser = {
  headers(): Promise<Record<string, string>>   // client header and authorization; for NestJS also the session binding header
  proofToken?(nonce: string): Promise<string>  // NestJS only: the Auth0 ID token the proof step sends
}
export type FilesState = { who: FilesUser; keys: { snapshot: string; export: string; upload: string }; challenge: { challenge_id: string; nonce: string }; proofId: string; snapshotId: string; exportId: string; importId: string; wallet: string; food: string; transaction: string }
export type FileStep = { name: string; signIn?: true; build(st: FilesState): Promise<Request>; after?(st: FilesState, body: unknown): void }
export const STEPS: FileStep[]                 // 42 steps; signIn on 'export: challenge' and 'export: proof'
export function runFilesSession(send: Send, who: FilesUser, mode: 'nest' | 'vault'): Promise<(Seen | null)[]>
export function nestUser(edge: Edge): Promise<FilesUser>                                // edge.member(): access token and x-gogospend-session, with proofToken
export function vaultUser(vault: { send: Send }, issuer: JwksFixture): Promise<FilesUser> // a fresh { sub, sid } that accepted terms v1 on the vault; tokens carry sid and auth_time
export function vaultHeaders(issuer: JwksFixture): Promise<Record<string, string>>       // one request's headers for a fresh { sub, sid }, with content-type JSON

// test/support/vault-worker.ts
export type VaultWorker = DevWorker & { send: Send; config: string }   // config: the file wrangler dev started
export function startVaultWorker(o: { issuer: JwksFixture; policies: unknown; vars?: Record<string, string>; background?: 'on' | 'off' }): Promise<VaultWorker>
```

- In `vault` mode the two sign-in steps are not sent (their entry is `null`) and `st.proofId` is a fresh UUID (B8, B12). The steps never sign a token: each request's headers come from `st.who.headers()`, and the proof step's ID token from `st.who.proofToken` (it throws when the user has none).
- `background: 'off'` writes `dist/jotbun_vault/wrangler.parity.json` next to the built config (so its `main` still resolves): a copy without `workflows`, `containers` and `queues.consumers`; the parse queue's producer stays. `vars` are passed as `--var` after the issuer's settings.
