## Decisions the reviewer should check first

1. **`rageClickConfig: { enabled: false }`**, beyond the plan's exact options object. posthog-ios detects rage taps and sends `$rageclick` from the native SDK, so outside `before_send` (the SDK's own option docs say so of the other native event it sends, `$push_notification_opened`; the rage-click option is documented as native-only and on by default), whenever the native SDK runs; crash reports without consent start it before any yes (the plugin's `setup()` runs for native crash capture), and on the keypad a repeated tap is an amount. The plan's options alone would break "nothing analytic is sent before yes" and "never tap positions". Android has no native rage-click detection. If the controller rules otherwise, drop the line, its expectation in `posthogOptions.test.ts` and mutant M7.
2. **1D-T1's follow-ups in `controller.ts`** (the controller's assignment for this brief gave them to 1D-T6; the plan's "1D-T1's files stay as built" is about the foreground test's harness):
   - `AccountChangedError` (exported, same message) for a yes refused because the account changed; `a yes waiting behind another call is refused once another account has signed in` now asserts the class.
   - The gap is closed: `answer('no')` writes `{ answer: 'no', pendingServer: true }` at the call, and the queued work awaits that same write before its own write and the server call (1D-T1's receipt gives this design). A no given while a refresh or another answer is on its way now survives the app being closed. New tests: `a no given while a refresh is on its way is stored at once`, `the no stored at the call lands before the queue writes and sends it`.
   - Same-account yes, documented and pinned: a queued yes is still sent when, by the time the queue reaches it, the account that gave it has signed out and in again (it is that account's answer). New test: `a yes waiting while the same account signs out and in again is still sent for it`.
   - Not closed: a no not yet delivered is still lost when another account signs in on the same phone first (one stored answer per phone; per-account storage changes 1D-T1's store interface), or when the explicit sign-out erase runs (plan Step 4). Recorded as a risk.
   - Callers catch every rejection: `AnalyticsUpkeep` catches `setAccount` and `refresh`; `answer` is 1D-T8's (the JSDoc says when it rejects).
3. **The session-change signal is R86's epoch.** On each session notification `AnalyticsUpkeep` reads `getV1SessionSnapshot()`: no binding, `setAccount(null)`; a binding whose epoch differs from the last one seen, `setAccount(null)` **at once**, then `fetchAccountId()`, then `setAccount(id)` if the epoch is still the same. The first session it sees is not a change (no `setAccount(null)`), which keeps the plan's "a session already restored at start" expectation of one reset, then one identify. Within one epoch the account is read once: a foreground re-reads `/v1/me` only when that read failed, then always calls `refresh()`. A newer call or a moved epoch discards an older read.
4. The `/v1/me/analytics` answer and the privacy-preferences answer are validated as exact objects (no extra keys), as the transport's validators are; a fallback happens only on a 404 `NOT_FOUND`.
5. `appAnalytics` makes the shared transport at the first server call, not at install (the root layout's render test cannot load the real `sharedV1`), and takes `app_version` and `platform` from `v1ClientVersion()`. `posthogClient` drops refusals from the promise-returning `screen` and replay calls.

## Appendix B: `apps/app/src/analytics/analyticsServer.ts`

```ts
// The consent and identity calls over the v1 transport (plan 1D-T6), pure.
// The server's answer is read from GET /v1/me/analytics (1D-T11); a server
// without that route answers 404 NOT_FOUND, and then /v1/me says only whether
// analytics is on, with no analytics id. The answer is sent with
// PUT /v1/me/privacy-preferences, and the server must say it back.
import { V1Error } from '../api/client.ts';
import type { V1Transport } from '../api/client.ts';
import type { AnalyticsDeps, ServerConsent } from './controller.ts';

type AnalyticsAnswer = { analytics_id: string | null; weeks_since_signup: number; consent: ServerConsent };

const ANALYTICS_ID = /^[0-9a-f]{64}$/;
const MAX_WEEKS = 1040;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function hasExactly(value: Record<string, unknown>, keys: readonly string[]): boolean {
  return Object.keys(value).sort().join(',') === [...keys].sort().join(',');
}

function isAnalyticsAnswer(value: unknown): value is AnalyticsAnswer {
  if (!isRecord(value) || !hasExactly(value, ['analytics_id', 'weeks_since_signup', 'consent'])) return false;
  const id = value.analytics_id;
  const weeks = value.weeks_since_signup;
  const consent = value.consent;
  return (id === null || (typeof id === 'string' && ANALYTICS_ID.test(id)))
    && typeof weeks === 'number' && Number.isInteger(weeks) && weeks >= 0 && weeks <= MAX_WEEKS
    && (consent === 'yes' || consent === 'no' || consent === 'unasked');
}

function isMeWithPrivacy(value: unknown): value is { privacy: { analytics: boolean } } {
  return isRecord(value) && isRecord(value.privacy) && typeof value.privacy.analytics === 'boolean';
}

/** A server that predates 1D-T11: the route table has no /v1/me/analytics. */
function routeMissing(error: unknown): boolean {
  return error instanceof V1Error && error.status === 404 && error.code === 'NOT_FOUND';
}

export function createAnalyticsServer(transport: V1Transport): AnalyticsDeps['server'] {
  return {
    async read() {
      try {
        const answer = await transport.request('GET', '/v1/me/analytics', { validate: isAnalyticsAnswer });
        return { consent: answer.consent, analyticsId: answer.analytics_id, weeksSinceSignup: answer.weeks_since_signup };
      } catch (error) {
        if (!routeMissing(error)) throw error;
      }
      const me = await transport.request('GET', '/v1/me', { validate: isMeWithPrivacy });
      return { consent: me.privacy.analytics ? 'yes' : 'unasked', analyticsId: null, weeksSinceSignup: null };
    },
    async setConsent(analytics) {
      await transport.request('PUT', '/v1/me/privacy-preferences', {
        body: { analytics },
        validate: (value): value is { analytics: boolean } => isRecord(value) && hasExactly(value, ['analytics']) && value.analytics === analytics,
      });
    },
  };
}
```

```ts
// This phone's analytics answer (plan 1D-T6): one value in the secure store on
// iOS and Android, readable only on this device while it is unlocked. The web
// keeps nothing. A read never throws, and anything that is not exactly a
// stored answer reads as none, because a failing read would stop a refresh
// before it reads the server's answer (1D-T1's notes).
import { Platform } from 'react-native';
import { isUuid } from '@gogospend/contracts';

import type { StoredConsent } from './controller';

export const ANALYTICS_CONSENT_KEY = 'gogospend.analytics-consent';

async function secureStore() {
  const store = await import('expo-secure-store');
  return { store, options: { keychainAccessible: store.WHEN_UNLOCKED_THIS_DEVICE_ONLY } };
}

function storedConsent(text: string | null): StoredConsent | null {
  if (text === null) return null;
  let value: unknown;
  try {
    value = JSON.parse(text);
  } catch {
    return null;
  }
  if (typeof value !== 'object' || value === null || Array.isArray(value)) return null;
  if (Object.keys(value).sort().join(',') !== 'accountId,answer,pendingServer') return null;
  const { accountId, answer, pendingServer } = value as Record<string, unknown>;
  if (!isUuid(accountId)) return null;
  if (answer !== 'yes' && answer !== 'no') return null;
  if (typeof pendingServer !== 'boolean') return null;
  return { accountId, answer, pendingServer };
}

export async function readStoredConsent(): Promise<StoredConsent | null> {
  if (Platform.OS === 'web') return null;
  try {
    const { store, options } = await secureStore();
    return storedConsent(await store.getItemAsync(ANALYTICS_CONSENT_KEY, options));
  } catch {
    return null;
  }
}

export async function writeStoredConsent(value: StoredConsent): Promise<void> {
  if (Platform.OS === 'web') return;
  const { store, options } = await secureStore();
  const { accountId, answer, pendingServer } = value;
  await store.setItemAsync(ANALYTICS_CONSENT_KEY, JSON.stringify({ accountId, answer, pendingServer }), options);
}

/** Forgets the answer (the explicit sign-out erase, with the diary). */
export async function eraseStoredConsent(): Promise<void> {
  if (Platform.OS === 'web') return;
  const { store, options } = await secureStore();
  await store.deleteItemAsync(ANALYTICS_CONSENT_KEY, options);
}
```

## Appendix F: `apps/app/src/analytics/AnalyticsUpkeep.tsx`

```tsx
// Analytics upkeep for the whole app (plan 1D-T6), rendered once in the root
// layout. It installs the controller, sends each screen as its route pattern
// (`useSegments` gives `["spend", "record", "[id]"]`, never the values),
// keeps the language current, follows the session's account, and reads the
// answer again on every return to the foreground, so a "no" given on another
// phone applies here. Every failure is caught: the next session change or
// foreground tries again.
import { useEffect, useState } from 'react';
import { AppState } from 'react-native';
import { useSegments } from 'expo-router';

import { fetchAccountId } from '@mobile/api/sharedV1';
import { getV1SessionSnapshot, subscribeSession } from '@mobile/auth/session';
import { useActiveLocale } from '@mobile/i18n/useCopy';
import { installAppAnalytics } from './appAnalytics';
import type { Analytics } from './controller';

const ignore = () => undefined;

/**
 * Sets the controller's account from the session. With no session, none. A
 * session other than the one last seen (its epoch moved: a sign-in or an
 * account switch) clears the account at once, before the new one is read, so a
 * yes still queued for the previous account cannot be sent under the new
 * session (1D-T1's review). Within one session the account is read once; a
 * failed read leaves it unset until the next call. Rejects when a read or the
 * controller fails.
 */
function accountFollower(analytics: Analytics): () => Promise<void> {
  // The epoch of the last session seen with a binding, and the epoch whose account the controller holds.
  let seenEpoch: number | null = null;
  let heldEpoch: number | null = null;
  let latest = 0;
  return async () => {
    const call = ++latest;
    const { binding, epoch } = getV1SessionSnapshot();
    if (binding === null) {
      seenEpoch = null;
      heldEpoch = null;
      await analytics.setAccount(null);
      return;
    }
    if (heldEpoch === epoch) return;
    if (seenEpoch !== null && seenEpoch !== epoch) {
      heldEpoch = null;
      await analytics.setAccount(null);
    }
    seenEpoch = epoch;
    const accountId = await fetchAccountId();
    const now = getV1SessionSnapshot();
    // A newer call, or a session that moved meanwhile, decides instead.
    if (call !== latest || now.epoch !== epoch || now.binding === null) return;
    heldEpoch = epoch;
    await analytics.setAccount(accountId);
  };
}

  useEffect(() => {
    let live = true;
    installAppAnalytics().then((installed) => { if (live) setInstalled(installed); }, ignore);
    return () => { live = false; };
  }, []);

  useEffect(() => {
    analytics?.setCommon({ language: locale });
  }, [analytics, locale]);

  useEffect(() => {
    analytics?.screen(JSON.parse(route) as string[]);
  }, [analytics, route]);

  useEffect(() => {
    if (!analytics) return;
    const follow = accountFollower(analytics);
    const unsubscribe = subscribeSession(() => { follow().catch(ignore); });
    // A session restored before this subscribed sends no notification.
    follow().catch(ignore);
    const subscription = AppState.addEventListener('change', (state) => {
      if (state !== 'active') return;
      follow().catch(ignore).then(() => analytics.refresh()).catch(ignore);
    });
    return () => {
      unsubscribe();
      subscription.remove();
    };
  }, [analytics]);

  return null;
}
```
