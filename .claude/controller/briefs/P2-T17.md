export type UndoPlan = { kind: 'remove_draft'; commandId: string } | { kind: 'delete_record'; recordId: string; version: 1 } | { kind: 'none' };

/** Each kind's own name in today's words, for a save with no merchant, note or known category. */
const KIND_NAMES: Readonly<Record<AddType, string>> = { expense: 'Expense', income: 'Income', transfer: 'Transfer' };

/** A landed save as the toast and the chip say it (`tc`: the app's words, for a built-in category's name). A transfer's walletId is its destination. */
export function noticeOf(saved: SavedEntry, lookups: Lookups, offline: boolean, tc: Translate): SavedNotice {
  const { draft } = saved;
  const categoryId = draft.type === 'transfer' ? null : draft.categoryByType[draft.type];
  const category = categoryId === null ? undefined : lookups.categories.get(categoryId);
  const title = draft.merchant.trim() || draft.note.trim() || (category === undefined ? '' : categoryName(category, tc)) || tc(KIND_NAMES[draft.type]);
  return {
    recordId: draft.recordId,
    commandId: draft.commandId,
    kind: draft.type,
    title,
    walletId: draft.type === 'transfer' ? draft.toWalletId : draft.walletId,
    via: saved.via,
    amountMinor: draft.amountMinor,
    offline,
  };
}

/** What Undo does for this save, from the outbox as it is now (see the note at the top). */
export function undoPlan(notice: SavedNotice, outbox: readonly OutboxCommand[]): UndoPlan {
  const command = outbox.find((each) => each.commandId === notice.commandId);
  if (command === undefined || command.state === 'acknowledged') return { kind: 'delete_record', recordId: notice.recordId, version: 1 };
  return canUndoDraft(command) ? { kind: 'remove_draft', commandId: command.commandId } : { kind: 'none' };
}

/** The confirmed total plus each expense command dated today whose record is not confirmed, unless Jotbun refused it. */
export function todayOutMinor(input: { summaryExpenseMinor: Minor; confirmedIds: ReadonlySet<string>; outbox: readonly OutboxCommand[]; today: string }): Minor {
  let total = BigInt(input.summaryExpenseMinor);
  for (const { payload, state } of input.outbox) {
    if (payload.type !== 'expense' || payload.occurred_on !== input.today) continue;
    if (state === 'needs_review' || input.confirmedIds.has(payload.id)) continue;
    total += BigInt(payload.amount_minor);
  }
  return total.toString();
}

/** The amount the toast draws: today's money out in the compact form, or the mask in private mode (W10, R65). */
export function toastAmount(todayOut: Minor, hidden: boolean): string {
  return hidden ? HIDDEN_AMOUNT : walletMoneyText(todayOut);
}

/** What the toast draws and says (`t` reads the saved lines and the shared ones). */
export function toastWords(notice: SavedNotice, todayOut: Minor | null, hidden: boolean, t: WalletT): { text: string; spoken: string } {
  if (notice.offline) {
    const text = t('saved.toastOffline');
    return { text, spoken: text };
  }
  if (todayOut === null) {
    const text = t('saved.toastPlain');
    return { text, spoken: text };
  }
  return {
    text: t('saved.toast', { amount: toastAmount(todayOut, hidden) }),
    spoken: t('saved.toast', { amount: hidden ? t('common.amountHidden') : spokenMoney(todayOut, t) }),
  };
}

function Toast({ id, notice, onGone }: { id: number; notice: SavedNotice; onGone(id: number): void }) {

  const t = useDraftCopy(LINES);
  const { hidden } = usePrivateMode();

  const diary = useSpendDiary();

  const keys = useCommandKeys();

  // The diary as the last commit left it, for the read that lands later.
  const latest = useRef(view);
  useEffect(() => {
    latest.current = view;
  });

  const [appeared, setAppeared] = useState(notice.offline);
  const [total, setTotal] = useState<Minor | null>(null);
  useEffect(() => {
    if (notice.offline) return undefined;

    const today = todayInBangkok(new Date());
    const timer = setTimeout(() => setAppeared(true), TOTAL_WAIT_MS);
    resources.history({ from: today, through: today, type: 'expense', sort: 'date_desc' }, undefined, HISTORY_MAX_LIMIT).then(
      (page) => {
        if (!live) return;
        // A day with more expenses than one page could leave a counted draft's record unlisted.
        if (page.next_cursor === null) {
          const now = latest.current;
          setTotal(todayOutMinor({
            summaryExpenseMinor: page.summary.expense_minor,
            confirmedIds: new Set(page.items.map((item) => item.id)),
            outbox: now.phase === 'ready' ? now.outbox : [],
            today,
          }));
        }
        setAppeared(true);
      },
      () => {
        if (live) setAppeared(true);
      },
    );
    return () => {
      live = false;
      clearTimeout(timer);
    };

  const plan = notice.via === 'device' && view.phase !== 'ready' ? NONE : undoPlan(notice, view.outbox);
  const words = toastWords(notice, total, hidden, t);
  const said = undo.phase === 'said' ? undo.text : null;

  const spoken = said ?? words.spoken;
  const offered = plan.kind !== 'none' && (undo.phase === 'ready' || (undo.phase === 'said' && undo.again));

  /** Undo, the rule (R61): one at a time, from the plan this render shows. */

// centre of the screen to its wallet's card on the gentle spring, shrinking and

export function SaveFlight({ notice }: { notice: SavedNotice | null }) {
  const reduced = useWalletReducedMotion();
  const { frameOf, measure } = useCardFrames();
  const [flight, setFlight] = useState<Flight | null>(null);
  useEffect(() => {
    const walletId = notice?.walletId ?? null;
    // Decided as the save lands, from the wallet's one answer (W6): under Reduce Motion nothing travels.
    if (notice === null || walletId === null || motionFor('save', reduced).motion.kind !== 'fly') return undefined;
    let live = true;
    const timers: ReturnType<typeof setTimeout>[] = [];

      timers.push(setTimeout(() => {
        if (live) run();
      }, ms));
    };
    after(TRAY_EXIT_MS, () => {
      measure();
      after(0, () => {
        const frame = frameOf(walletId);
        if (frame === null) return;

      });

      live = false;
      timers.forEach(clearTimeout);
    };
    // Once per notice: the host keys this part by its notice, and the flight is decided as the save lands.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [notice]);

  // The dependency array names the shared value itself (R72).

    opacity: 1 - progress.value * progress.value,
    transform: [{ translateX: dx * (1 - progress.value) }, { translateY: dy * (1 - progress.value) }, { scale: 1 - progress.value / 2 }],
  }), [progress, dx, dy]);
  return (
    <View
      {...DECORATIVE}
      testID="save-flight"
      pointerEvents="none"
      style={{ position: 'absolute', left: to.x - room / 2, top: to.y - CHIP_HALF, width: room, alignItems: 'center', zIndex: 20 }}
    >

        style={[

            minHeight: 2 * CHIP_HALF,
            maxWidth: room,
            flexDirection: 'row',
            alignItems: 'center',
