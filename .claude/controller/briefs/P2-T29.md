// Category command sender and state (plan P2-T29, moved out of CategoriesScreen.tsx).
// Sends category commands (create, update, archive, unarchive, setBudget) under
// their command keys, rebases drafts on acknowledgement, handles unconfirmed
// states, and formats conflicts and refusals.
import type { BudgetDto, CategoryDto } from '@gogospend/contracts';
import { useRef, useState } from 'react';

import { V1Error } from '@mobile/api/client';
import type { V1Outcome } from '@mobile/api/client';
import { useCopy } from '@mobile/i18n/useCopy';
import {
  SAVE_PROBLEM_WORDS, afterCategoryAck, budgetConflict, categoryConflict, commandSlot, readConflict,
} from '@mobile/spend/v1/redesign/planning/categoriesModel';
import type { CategoryCommand, CategoryDraft, ConflictView, FormErrors } from '@mobile/spend/v1/redesign/planning/categoriesModel';
import type { Translate } from '@mobile/spend/v1/redesign/shared';
import type { SpendV1Resources } from '@mobile/spend/v1/resources';
import { useSpendV1 } from '@mobile/spend/v1/SpendV1Context';
import type { SaveProblem } from '../../v1/parts';
import { isOwnCreate, ownCreateMatches, refusedBeforeProcessing, saveProblem, useCommandKeys, useMoneyLocale } from '../../v1/parts';

export type Origin = { success: string; draft: CategoryDraft | null; seen: BudgetDto | null };
export type CategoryCommandsStatus = 'idle' | 'saving' | 'unconfirmed';
export type CategoryConflict = { view: ConflictView; withYours: boolean };

/** Refusals that keep the shared words; those that speak of an entry are said in a category's (`SAVE_PROBLEM_WORDS`). */
const SHARED_WORDS: readonly V1Outcome[] = ['policy_required', 'signed_out', 'session_changed'];

/** `saveProblem` (unconfirmed, blocked, fields) in a category's words. */
export function categoryProblem(error: unknown, tc: Translate): SaveProblem {
  const problem = saveProblem(error, tc);
  const outcome = error instanceof V1Error ? error.outcome : undefined;
  const own = outcome ? SAVE_PROBLEM_WORDS[outcome] : undefined;
  if (own) return { ...problem, message: tc(own) };
  if ((outcome && SHARED_WORDS.includes(outcome)) || (outcome === 'invalid' && Object.keys(problem.fields).length > 0)) return problem;
  return { ...problem, message: tc('Could not save that. Try again.') };
}

/** Sends one command under its key. */
export function sendCommand(resources: SpendV1Resources, command: CategoryCommand, key: string) {
  if (command.slot === 'budget') return resources.setBudget(command.month, command.input, key, command.existing);
  switch (command.kind) {
    case 'create': return resources.createCategory(command.input, key);
    case 'update': return resources.updateCategory(command.id, command.version, command.changes, key);
    case 'archive': return resources.archiveCategory(command.id, command.version, key);
    case 'unarchive': return resources.unarchiveCategory(command.id, command.version, key);
  }

/** A refusal about the category's state (already archived, archived elsewhere, not archived): it changed elsewhere. */

  error instanceof V1Error && (error.outcome === 'not_found' || error.outcome === 'deleted' || 'archived' in (error.fields ?? {}));

export type CategoryCommandsOptions = {
  readonly onRebase?: (draft: CategoryDraft) => void;
  readonly onRefused?: (errors: FormErrors) => void;
  readonly onConflict?: (conflict: CategoryConflict) => void;
  readonly onStateChanged?: (message: string) => void;
  readonly onSuccess?: (origin: Origin) => void;
  readonly onRefresh?: () => void;
  readonly onReload?: () => void;
};

export function useCategoryCommands(options: CategoryCommandsOptions = {}) {
  const resources = useSpendV1();
  const tc = useCopy();
  const locale = useMoneyLocale();
  const keys = useCommandKeys();

  const [status, setStatus] = useState<CategoryCommandsStatus>('idle');
  const [message, setMessage] = useState<string | null>(null);
  const [resendBlocked, setResendBlocked] = useState(false);
  const [blocked, setBlocked] = useState(false);
  const [conflict, setConflict] = useState<CategoryConflict | null>(null);
  const [done, setDone] = useState<string | null>(null);
  const [unarchiving, setUnarchiving] = useState<string | null>(null);
  const pending = useRef<{ commands: readonly CategoryCommand[]; origin: Origin } | null>(null);
  const running = useRef(false);

  const locked = status !== 'idle';

  /**
   * Sends the commands in order. Each acknowledged category command rebases
   * the open draft, so if a later one is refused, saving again sends only
   * what is left. The first unconfirmed one freezes the screen with the
   * commands from it on, under their keys.
   */
  async function send(commands: readonly CategoryCommand[], origin: Origin): Promise<boolean> {
    if (running.current) return false;
    running.current = true;
    const resend = pending.current !== null;
    let draft = origin.draft;
    setStatus('saving');
    setMessage(null);
    setDone(null);
    try {
      for (const [index, command] of commands.entries()) {
        const slot = commandSlot(command);
        const key = keys.keyFor(slot, command);
        try {
          const ack = await sendCommand(resources, command, key);
          keys.settle(slot);
          draft = draft ? afterCategoryAck(draft, command, ack.version) : null;
          continue;
        } catch (error) {
          // The category's id is fixed for this draft: if it already exists, an earlier attempt committed.
          if (command.slot === 'category' && command.kind === 'create' && isOwnCreate(error, command.input.id) && ownCreateMatches(error, command.input)) {
            keys.settle(slot);
            const held = (error as V1Error).current as CategoryDto;
            draft = draft ? afterCategoryAck(draft, command, held.version) : null;
            continue;
          }
          const problem = categoryProblem(error, tc);
          // Once unconfirmed, a refusal before processing says nothing about that attempt: it stays frozen with its key.
          const stillUnknown = problem.unconfirmed || (resend && index === 0 && refusedBeforeProcessing(error));
          if (!stillUnknown) keys.settle(slot);
          const rebased = draft;
          if (rebased && options.onRebase) options.onRebase(rebased);
          if (stillUnknown) {
            pending.current = { commands: commands.slice(index), origin: { ...origin, draft } };
            setResendBlocked(problem.blocked);
            setMessage(problem.message);
            setStatus('unconfirmed');
            return false;
          }
          pending.current = null;
          setResendBlocked(false);
          setBlocked(problem.blocked);
          setStatus('idle');
          refused(error, command, problem, { ...origin, draft });
          return false;
        }
      }
      pending.current = null;
      setResendBlocked(false);
      setStatus('idle');
      setDone(origin.success);
      if (options.onSuccess) options.onSuccess(origin);
      if (options.onRefresh) options.onRefresh();
      return true;
    } finally {
      running.current = false;
      setUnarchiving(null);
    }
  }

  /** A definite refusal: a conflict to show, a change made elsewhere, or a problem to fix here. */
  function refused(error: unknown, command: CategoryCommand, problem: SaveProblem, origin: Origin) {
    const outcome = error instanceof V1Error ? error.outcome : undefined;
    if (outcome === 'conflict') {
      const copy = readConflict(command, (error as V1Error).current);
      if (copy?.kind === 'category') {
        const conf: CategoryConflict = { view: categoryConflict(origin.draft, copy.server, tc), withYours: origin.draft !== null };
        setConflict(conf);
        if (options.onConflict) options.onConflict(conf);
        return;
      }
      if (copy?.kind === 'budget' && origin.draft && origin.seen) {
        const conf: CategoryConflict = { view: budgetConflict(origin.draft, origin.seen, copy.server, { tc, locale }), withYours: true };
        setConflict(conf);
        if (options.onConflict) options.onConflict(conf);
        return;
      }
    }
    if (outcome === 'conflict' || stateChanged(error)) {
      // No usable copy of theirs: read everything again rather than loop on a stale version.
      const msg = tc('This category changed on another device.');
      setMessage(msg);
      if (options.onStateChanged) options.onStateChanged(msg);
      if (options.onReload) options.onReload();
      return;
    }
    setMessage(problem.message);
    const fields = problem.fields;
    const errors: FormErrors = { label: fields.label, limit: fields.category_limits ?? fields.overall_limit_minor };
    if (options.onRefused) options.onRefused(errors);
  }

  function tryAgain() {
    const stored = pending.current;
    if (stored && !resendBlocked) void send(stored.commands, stored.origin);
  }

  function reload() {
    setConflict(null);
    setMessage(null);
    if (options.onReload) options.onReload();
  }

  function reset() {
    pending.current = null;
    setResendBlocked(false);
    setBlocked(false);
    setStatus('idle');
    setMessage(null);
    setConflict(null);
    setDone(null);
    setUnarchiving(null);
  }

  return {
    status,
    message,
    resendBlocked,
    blocked,
    conflict,
    done,
    unarchiving,
    locked,
    send,
    tryAgain,
    reload,
    reset,
    setMessage,
    setDone,
    setConflict,
    setUnarchiving,
  };
}

// - 12 glyphs at 48 pt (CATEGORY_PICKER_GLYPHS)

import { CATEGORY_COLORS } from '@gogospend/contracts';
import { useEffect, useMemo, useState } from 'react';

const ALL_COPY = { ...COMMON_COPY, ...COPY } as const;

  'swatch.peach',
  'swatch.sky',
  'swatch.light_violet',

export function CategoryTray({ visible, onClose, category, type: initialType, now: explicitNow }: CategoryTrayProps) {
  const resources = useSpendV1();
  const tc = useCopy();
  const t = useDraftCopy(ALL_COPY);
  const locale = useMoneyLocale();
  const activeLocale = useActiveLocale();
  const colors = useWalletColors();
  const { iconOf, setIcon } = useCategoryIcons();

  const now = explicitNow ?? new Date();
  const month = monthOf(todayInBangkok(now));
  const budgetLoad = useLoad(async () => resources.budget(month), [resources, month]);
  const budget: BudgetDto | null = budgetLoad.state === 'ready' ? budgetLoad.data : null;

  const isNew = category === null;
  const isBuiltin = category?.builtin_key !== null && category?.builtin_key !== undefined;

  const [categoryType, setCategoryType] = useState<CategoryType>(category?.type ?? initialType);
  const [name, setName] = useState<string>('');
  const [pickedGlyph, setPickedGlyph] = useState<GlyphKey | null>(null);
  const [pickedColor, setPickedColor] = useState<string>(TRAY_SWATCHES[0]);
  const [hasLimit, setHasLimit] = useState(false);
  const [limitText, setLimitText] = useState('');
  const [saving, setSaving] = useState(false);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);

  const commands = useCategoryCommands();

  // Reset state when category or visible changes

    if (visible) {
      if (category) {
        setCategoryType(category.type);
        setName(category.label ?? '');
        setPickedColor(category.color ?? TRAY_SWATCHES[0]);
        setPickedGlyph(iconOf(category));
        const currentLimit = budget?.category_limits.find((l) => l.category_id === category.id)?.limit_minor;
        if (currentLimit !== undefined && currentLimit !== null) {
          setHasLimit(true);
          const satang = BigInt(currentLimit);
          const baht = satang / 100n;
          const remainder = satang % 100n;
          setLimitText(remainder === 0n ? baht.toString() : (Number(satang) / 100).toFixed(2));
        } else {

        setCategoryType(initialType);
        setName('');
        setPickedColor(TRAY_SWATCHES[0]);

    if (!check.ok) {

          total: walletMoneyText(check.total),
          budget: walletMoneyText(check.overall),

    setErrorMsg(null);

      const generatedId = isNew ? newId() : category.id;

      const draft = isNew

            ...newCategoryDraft({ id: generatedId, type: categoryType, color: pickedColor }),
            label: name,
            limitText: effectiveLimit ? limitText : '',
            baseLimit: null,
          }
        : {
            ...editCategoryDraft(category, budget),
            label: name,
            color: pickedColor,
            limitText: effectiveLimit ? limitText : '',
          };

      const plan = planSave(draft, { budget, tc, locale });
      if (!plan.ok) {
        setErrorMsg(plan.errors.label ?? plan.errors.limit ?? 'Error saving');
        setSaving(false);
        return;
      }

      const ok = await commands.send(plan.commands, {
        success: isNew ? tc('Created.') : tc('Saved.'),
        draft,
        seen: budget,
      });

      if (ok) {
        // The icon is kept for the account only after Jotbun accepted the category
        if (pickedGlyph) {
          try {
            await setIcon(generatedId, pickedGlyph);
          } catch {
            // Keep no failure outside account
          }
        }
        onClose();
      } else {
        setErrorMsg(commands.message ?? 'Could not save');
      }
    } catch {
      setErrorMsg('Could not save that. Try again.');
    } finally {
      setSaving(false);
    }

              value={name}
              onChangeText={setName}

              style={[styles.textInput, { color: colors.ink, borderColor: colors.hairline, backgroundColor: colors.card }]}

        {isNew && (
          <View role="radiogroup" aria-label={t('categories.title')} style={[styles.typeSwitch, { backgroundColor: colors.raised }]}>
            <Pressable
              role="radio"
              aria-checked={categoryType === 'expense'}
              aria-label={t('categories.expense')}
              onPress={() => setCategoryType('expense')}
              style={[styles.typeSegment, categoryType === 'expense' && [styles.typeSegmentActive, { backgroundColor: colors.card }]]}
            >
              <WalletText role="label" style={{ color: categoryType === 'expense' ? colors.ink : colors.quiet, fontWeight: categoryType === 'expense' ? '700' : '400' }}>
                {t('categories.expense')}
              </WalletText>
            </Pressable>
            <Pressable
              role="radio"
              aria-checked={categoryType === 'income'}
              aria-label={t('categories.income')}
              onPress={() => setCategoryType('income')}
              style={[styles.typeSegment, categoryType === 'income' && [styles.typeSegmentActive, { backgroundColor: colors.card }]]}

                {t('categories.income')}

            </Pressable>
          </View>
        )}

        {/* 12 Glyphs at 48 pt */}
        <View style={styles.section}>
          <WalletText role="label" style={styles.sectionLabel}>{t('categories.icon')}</WalletText>

// Shows C3's nine everyday categories with checkboxes. Picked missing items
// are created at once with packCommands; categories the person already has
// are disabled and marked with "มีแล้ว". Pack categories show their glyph

import { useSpendV1 } from '@mobile/spend/v1/SpendV1Context';

import { useDraftCopy } from '../copy/useDraftCopy';
import { COPY } from './copy';

export type ThaiPackTrayProps = {
  visible: boolean;
  onClose(): void;
  categories: readonly CategoryDto[];
};

export function ThaiPackTray({ visible, onClose, categories }: ThaiPackTrayProps) {
  const resources = useSpendV1();
  const tc = useCopy();
  const t = useDraftCopy(COPY);
  const activeLocale = useActiveLocale();
  const colors = useWalletColors();

  const now = new Date();
  const month = monthOf(todayInBangkok(now));
  const budgetLoad = useLoad(async () => resources.budget(month), [resources, month]);
  const budget = budgetLoad.state === 'ready' ? budgetLoad.data : null;

  const missing = useMemo(() => missingPackItems(categories), [categories]);
  const missingKeys = useMemo(() => new Set(missing.map((item) => item.key)), [missing]);

  const [picked, setPicked] = useState<Set<string>>(() => new Set(missingKeys));
  const [saving, setSaving] = useState(false);

  const commands = useCategoryCommands();

  useEffect(() => {
    if (visible) {
      setPicked(new Set());
      setSaving(false);
    }
  }, [visible, missingKeys]);

  function togglePicked(key: string) {
    if (!missingKeys.has(key)) return;
    setPicked((current) => {
      const next = new Set(current);
      if (next.has(key)) next.delete(key);
      else next.add(key);
      return next;
    });
  }

  async function handleAdd() {
    const keysToAdd = Array.from(picked).filter((k) => missingKeys.has(k));
    if (keysToAdd.length === 0 || saving) return;
    setSaving(true);
    try {
      const locale = activeLocale === 'th' ? 'th' : 'en';
      const cmds = packCommands(categories, locale, newId, { only: keysToAdd });
      const ok = await commands.send(cmds, {
        success: tc('Added.'),
        draft: null,
        seen: budget,
      });
      if (ok) {
        // pack categories show their glyph through iconFor, so nothing is kept for them
        onClose();
      }
    } finally {
      setSaving(false);
    }
  }

  const pickedCount = Array.from(picked).filter((k) => missingKeys.has(k)).length;

  return (
    <Tray visible={visible} onClose={onClose} title={t('categories.packHeading')}>
      <View style={styles.container}>
        <WalletText role="body" style={[styles.intro, { color: colors.quiet }]}>

            const hasItem = !missingKeys.has(item.key);
            const isChecked = hasItem || picked.has(item.key);
            const glyph = PACK_GLYPH[item.key] ?? 'receipt';
            const name = activeLocale === 'th' ? item.th : item.en;
            const lineKey = `pack.${item.key}` as keyof typeof COPY;
            const line = COPY[lineKey] ? t(lineKey) : '';

            return (
              <Pressable
                key={item.key}
                role="checkbox"
                aria-label={name}
                aria-checked={isChecked}
                aria-disabled={hasItem}
                disabled={hasItem || saving}
                onPress={() => togglePicked(item.key)}
                style={[styles.row, { borderBottomColor: colors.hairline }]}
              >
                <View style={[styles.glyphTile, { backgroundColor: item.color }]}>
                  <CategoryGlyph glyph={glyph} size={20} color={colors.ink} />
                </View>

                <View style={styles.rowDetails}>
                  <WalletText style={[styles.itemName, hasItem && { color: colors.quiet }]}>
                    {name}
                  </WalletText>
                  <WalletText role="caption" style={[styles.itemLine, { color: colors.quiet }]}>
                    {line}
                  </WalletText>
                </View>

                {hasItem ? (
                  <View style={[styles.haveBadge, { backgroundColor: colors.raised }]}>
                    <WalletText role="caption" style={[styles.haveText, { color: colors.quiet }]}>
                      {t('categories.packHave')}
                    </WalletText>
                  </View>
                ) : (
                  <View
                    style={[
                      styles.checkboxBox,
                      {
                        backgroundColor: isChecked ? colors.ink : 'transparent',
                        borderColor: isChecked ? colors.ink : colors.quiet,
                      },
                    ]}
                  >
                    {isChecked ? (
                      <WalletText style={{ color: colors.card, fontSize: 13, fontWeight: '700' }}>✓</WalletText>
                    ) : null}
                  </View>
                )}
              </Pressable>
            );
          })}
        </View>

        {/* Note box */}
        <View style={[styles.noteBox, { backgroundColor: colors.raised }]}>
          <WalletText role="caption" style={[styles.noteText, { color: colors.ink }]}>
            {t('categories.packNote')}
          </WalletText>
        </View>

async function ready() {

function startAnimations() {
  act(() => {
    for (const node of document.querySelectorAll<HTMLElement>('[style*="visibility: hidden"]')) {
      node.dispatchEvent(new Event('animationstart'));
    }
  });
}

function named(role: string, name: string | RegExp, inside: HTMLElement = document.body): HTMLElement {
  const elements = within(inside).queryAllByRole(role, { name });
  expect(elements.length, `a ${role} named ${String(name)}`).toBeGreaterThan(0);
  return elements[0] as HTMLElement;
}

function found(testID: string): HTMLElement {

}

function shown(words: string | RegExp, inside: HTMLElement = document.body): HTMLElement {

  expect(elements.length, String(words)).toBeGreaterThan(0);
  return elements[0] as HTMLElement;
}

  return (
    <SpendV1Provider resources={getSharedV1Resources()}>
      <CategoriesScreen />
    </SpendV1Provider>
  );
}
