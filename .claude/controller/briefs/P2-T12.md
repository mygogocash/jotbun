```tsx
// Where each wallet's card sits on screen (P2-T12), for the saved moment's
// flying chip (P2-T17: it "travels from the bottom centre to the card"). A
// card's view registers under its wallet's id through the ref that
// useRegisterCardFrame gives, and frameOf answers with where that view was
// last measured in the window (measureInWindow), or null while no card of
// that wallet is on screen, or before it has been measured. A view is
// measured when it registers, which is after layout, and again whenever
// measure() asks; CardStack asks after each of its own layouts, so a card is
// measured again where it now is when the stack moves within the page. Each
// answer replaces the last when it arrives (on the web a moment later, once
// the page has laid out).
//
// One view per wallet id: the last one registered. A view that leaves forgets
// its wallet's frame, never another view's, and a measurement that lands after
// its view has left is never given out. Outside a provider nothing is kept:
// every frame is null and measure() does nothing, so a card anywhere (the
// catalogue, a test) needs no provider. WalletShell mounts it (P2-T18).
import { createContext, useCallback, useContext, useRef, useState } from 'react';
import type { ReactNode } from 'react';
import type { HostInstance } from 'react-native';

export type CardFrame = { x: number; y: number; width: number; height: number };

export type CardFrames = {
  /** Where the wallet's card was last measured in the window; null while none is on screen, or before it is measured. */
  frameOf(walletId: string): CardFrame | null;
  /** Measures every registered card again; each new frame replaces the last when it arrives. */
  measure(): void;
};

type Registry = CardFrames & {
  register(walletId: string, view: HostInstance): void;
  forget(walletId: string, view: HostInstance): void;
};

function createRegistry(): Registry {
  const views = new Map<string, HostInstance>();
  // Keyed by the view, so a frame measured for a view that has left never answers for its wallet.
  const frames = new WeakMap<HostInstance, CardFrame>();
  const measureView = (view: HostInstance) => {
    view.measureInWindow((x, y, width, height) => {
      frames.set(view, { x, y, width, height });
    });
  };
  return {
    register(walletId, view) {
      views.set(walletId, view);
      measureView(view);
    },
    forget(walletId, view) {
      if (views.get(walletId) === view) views.delete(walletId);
    },
    frameOf(walletId) {
      const view = views.get(walletId);
      return view === undefined ? null : (frames.get(view) ?? null);
    },
    measure() {
      for (const view of views.values()) measureView(view);
    },
  };
}

const CardFramesContext = createContext<Registry | null>(null);

const NO_FRAMES: CardFrames = { frameOf: () => null, measure: () => undefined };

export function CardFramesProvider({ children }: { children: ReactNode }) {
  const [registry] = useState(createRegistry);
  return <CardFramesContext.Provider value={registry}>{children}</CardFramesContext.Provider>;
}

/** The ref for a wallet card's view: it registers the view under the wallet's id, and forgets it when the view leaves. */
export function useRegisterCardFrame(walletId: string): (view: HostInstance | null) => void {
  const registry = useContext(CardFramesContext);
  const registered = useRef<HostInstance | null>(null);
  return useCallback(
    (view: HostInstance | null) => {
      if (registry === null) return;
      if (registered.current !== null) registry.forget(walletId, registered.current);
      registered.current = view;
      if (view !== null) registry.register(walletId, view);
    },
    [registry, walletId],
  );
}

/** Where the wallet cards on screen are, and a way to measure them again. */
export function useCardFrames(): CardFrames {
  return useContext(CardFramesContext) ?? NO_FRAMES;
}
```

```tsx
    // ax-large-home: one column of spine cards, front card first, with no shadows.
    return (

        {[...wallets].reverse().map((wallet, index) => (
          <StackCard key={wallet.id} wallet={wallet} front={index === 0} variant="spine" frame={null} onOpen={onOpen} />
        ))}
      </View>
    );
  }
  return (

      {wallets.map((wallet, index) => (

      ))}
    </View>
  );

/** Where a card sits in the stack: the cards behind are placed STACK_PEEK apart over the front card, which is in the flow. */
function stackFrame(index: number, last: number, isDark: boolean): ViewStyle {
  const place: ViewStyle = index === last
    ? { marginTop: STACK_PEEK * index }
    : { position: 'absolute', top: STACK_PEEK * index, left: 0, right: 0 };
  // The first card has no card behind it to cast a shadow on.
  const shadow = isDark || index === 0 ? undefined : index === last ? SHADOW.front : SHADOW.back;
  return { ...place, borderRadius: CARD_RADIUS, boxShadow: shadow };
}

function StackCard({ wallet, front, variant, frame, onOpen }: {
  wallet: StackWallet;
  front: boolean;
  variant: 'stack' | 'spine';
  frame: ViewStyle | null;
  onOpen: (walletId: string) => void;
}) {
  const reduced = useWalletReducedMotion();
  const register = useRegisterCardFrame(wallet.id);
  const lift = useSharedValue(0);
  // The dependency array names the shared value itself (R72).
  const lifted = useAnimatedStyle(() => ({ transform: [{ translateY: lift.value }] }), [lift]);

  const open = () => {
    const plan = motionFor('walletOpen', reduced);
    if (plan.haptic !== null) playHaptic(plan.haptic);
    if (plan.motion.kind === 'spring') {
      const spring = { ...motion.spring[plan.motion.spring], reduceMotion: ReduceMotion.Never };
      lift.value = withSequence(withSpring(-CARD_LIFT, spring), withSpring(0, spring));
    }
    onOpen(wallet.id);
  };

  return (
    <Animated.View testID={`card-stack-${wallet.id}`} style={[frame, lifted]}>
      <View ref={register} collapsable={false}>
        <WalletCard wallet={front ? wallet : { ...wallet, todayOutMinor: null }} look={wallet.look} variant={variant} onPress={open} />
      </View>

  );
}
```

```tsx
/** How far a card is lifted, in points (0 while it is in its place). */
function liftOf(id: string): number {
  const transform = cardFrame(id).style.transform;
  return transform === '' ? 0 : Number.parseFloat(transform.slice('translateY('.length));
}

/** Each card's lift, frame by frame (16 ms), for `ms`, on the faked timers. */
function trace(ids: readonly string[], ms: number): Record<string, number[]> {
  const lifts: Record<string, number[]> = Object.fromEntries(ids.map((id) => [id, []]));
  for (let at = 0; at < ms; at += 16) {
    act(() => { vi.advanceTimersByTime(16); });
    for (const id of ids) lifts[id].push(liftOf(id));
  }
  return lifts;
}

type Place = { left: number; top: number; width: number; height: number };

/** Gives elements a place in the window, as the page's layout would; anything else is at 0, 0 with no size, as happy-dom says. */
function placeViews(places: Map<Element, Place>) {
  vi.spyOn(HTMLElement.prototype, 'getBoundingClientRect').mockImplementation(function placed(this: HTMLElement) {
    const { left, top, width, height } = places.get(this) ?? { left: 0, top: 0, width: 0, height: 0 };
    return { left, top, width, height, x: left, y: top, right: left + width, bottom: top + height, toJSON: () => ({}) } as DOMRect;
  });
}

/** What a layout of this view tells its onLayout (see the note at the top). */
function layOut(element: HTMLElement) {
  const handler = (element as unknown as { __reactLayoutHandler?: (event: unknown) => void }).__reactLayoutHandler;
  expect(typeof handler, 'the view has an onLayout').toBe('function');
  act(() => { handler?.({ nativeEvent: { layout: { x: 0, y: 0, width: 370, height: 362 } }, timeStamp: 0 }); });
}

  const places = new Map<Element, Place>();

  const frameOf = (id: string): CardFrame | null => (frames as CardFrames | null)?.frameOf(id) ?? null;

  const expected = (down: number) => HOME.map((_, index) => ({ x: 16, y: 300 + down + STACK_PEEK * index, width: 370, height: 200 }));

  // Nothing is known until the page has laid out.

  placeAt(-100);
  act(() => { (frames as CardFrames | null)?.measure(); });
  await settle();
  expect(HOME.map(({ id }) => frameOf(id))).toEqual(expected(-100));

  // A card that leaves is forgotten at once, and a measurement already asked for it is never given out; the others keep theirs.
  placeAt(0);

  placeList(25);
  layOut(stack());
  await settle();
  expect(frameOf('w-credit')).toEqual({ x: 16, y: 145, width: 370, height: 280 });
  listView.unmount();
  expect(frameOf('w-credit')).toBeNull();
});

test('one view per wallet id: the last registered answers, and a view that leaves never forgets another; outside a provider nothing is kept', async () => {

  const frameOf = (id: string): CardFrame | null => (frames as CardFrames | null)?.frameOf(id) ?? null;

    <CardFramesProvider>
      <FramesProbe onFrames={take} />
      {children}
    </CardFramesProvider>
  );
  const A = { x: 10, y: 20, width: 30, height: 40 };
  const B = { x: 50, y: 60, width: 70, height: 80 };

  const view = mountWallet(tree([<Registrar key="a" walletId="w1" testID="a" />]));
```
