```ts
  return value === null ? {} : { category_key: value };
}

  if (!Number.isFinite(value)) return 0;

}
```

```ts
// The draft review (P3-T5): how a draft Bun made fared once the person saved
// or dropped it, sent as one ai_draft_reviewed. Pure, so Node tests run it.
//
// - A draft's fields hold an amount, a merchant and a note, but fieldsEdited
//   turns them into a count of the fields that changed, and that count, the
//   outcome, the model and escalated are all the event carries (§7).
// - '' and null are the same: an empty field is empty however the form holds it.
// - The tally (a development build's count, src/dev/draftTally.ts) comes after
//   the event, and a tally that fails is dropped: a count never fails a review.
import { trackAiDraftReviewed } from '../../../analytics/events.ts';
import type { AiModelName, DraftOutcome } from '../../../analytics/events.ts';

export type DraftFields = { type: string; amountMinor: string | null; occurredOn: string | null; categoryId: string | null; walletId: string | null; merchant: string | null; note: string | null };

const FIELDS: readonly (keyof DraftFields)[] = ['type', 'amountMinor', 'occurredOn', 'categoryId', 'walletId', 'merchant', 'note'];

function same(before: string | null, after: string | null): boolean {
  return (before ?? '') === (after ?? '');
}

export function fieldsEdited(before: DraftFields, after: DraftFields): number {
  return FIELDS.filter((field) => !same(before[field], after[field])).length;
}

export function savedOutcome(before: DraftFields, after: DraftFields): { outcome: 'kept' | 'edited'; fieldsEdited: number } {
  const edited = fieldsEdited(before, after);
  return { outcome: edited === 0 ? 'kept' : 'edited', fieldsEdited: edited };
}

export function reviewDraft(
  p: { outcome: DraftOutcome; fieldsEdited: number; model: AiModelName; escalated: boolean },
  tally?: (outcome: DraftOutcome, model: AiModelName) => void | Promise<void>,
): void {
  trackAiDraftReviewed(p);
  if (tally === undefined) return;
  void Promise.resolve(tally(p.outcome, p.model)).catch(() => undefined);
}
```

```ts
// useDraftReview (P3-T5): the review a screen calls once a draft Bun made is
// saved or dropped (P3-T13 and P3-T18 call it). Each review sends one
// ai_draft_reviewed and, in a development build only, counts its outcome in
// the draft tally for the phase 3 proof.
//
// - The tally is reached only through the conditional below. In a release
//   build babel-preset-expo inlines __DEV__ as false and Metro folds the
//   conditional to undefined before it collects dependencies, so the import()
//   goes with it and src/dev/draftTally.ts never reaches a release bundle
//   (P3-T22 adds an export check that fails a bundle holding it). `typeof`
//   keeps this module loadable where nothing defines __DEV__, as in Node
//   tests.
// - The tally loads with import(), not require(): the render suite is a
//   development build whose require() is Node's own, which cannot resolve
//   @mobile or read TypeScript, while import() goes through the bundler there
//   as it does in the app.
// - reviewDraft drops a count that fails, so a review never fails on it.
import type { AiModelName, DraftOutcome } from '../../../analytics/events.ts';
import { reviewDraft } from './draftReview.ts';

type Review = Parameters<typeof reviewDraft>[0];

const tally: ((outcome: DraftOutcome, model: AiModelName) => Promise<void>) | undefined =
  typeof __DEV__ !== 'undefined' && __DEV__
    ? (outcome, model) => import('@mobile/dev/draftTally').then((dev) => dev.addDraftOutcome(outcome, model))
    : undefined;

function review(p: Review): void {
  reviewDraft(p, tally);
}

/** The same function every time, so a screen may keep it in a hook's dependency list. */
export function useDraftReview(): (p: Review) => void {
  return review;
}
```

`apps/app/src/dev/draftTally.ts`:
```ts
// The draft tally (P3-T5), for development builds only: how many of Bun's
// drafts were kept, edited or dropped, per model, so the phase 3 proof can
// tell the AI's share from the rules'. useDraftReview reaches it behind
// __DEV__, so no release bundle holds it, and P3-T22 adds an export check
// that fails a bundle holding DRAFT_TALLY_KEY. It keeps counts and nothing
// else: no amount, note, merchant or other words from a draft.
//
// - Every read and write goes through one queue, in call order, so drafts
//   reviewed together (P3-T18 saves several at once) each count, and a read
//   sees every count made before it. A failed write rejects its own call
//   only; the queue goes on.
// - A stored value that is not the tally's shape reads as zeros, and a count
//   that is not a whole number from 0 up reads as 0. A write keeps the three
//   models and their three counts only.
// - AsyncStorage loads only when no backend is passed, so Node tests load this
//   module without React Native.
import type { AiModelName, DraftOutcome } from '../analytics/events.ts';
import type { StoreBackend } from '../wallet/store/walletStore.ts';

export const DRAFT_TALLY_KEY = 'jotbun.dev.draft-tally';

export type DraftCounts = { kept: number; edited: number; dropped: number };
export type DraftTally = Readonly<Record<AiModelName, DraftCounts>>;

type TallyStore = Pick<StoreBackend, 'getItem' | 'setItem' | 'removeItem'>;

let queue: Promise<unknown> = Promise.resolve();

function serially<T>(work: () => Promise<T>): Promise<T> {
  const run = queue.then(work);
  queue = run.catch(() => undefined);
  return run;
}

async function storeOf(backend: StoreBackend | undefined): Promise<TallyStore> {
  if (backend !== undefined) return backend;
  return (await import('@react-native-async-storage/async-storage')).default;
}

function fieldOf(value: unknown, key: string): unknown {
  return typeof value === 'object' && value !== null ? (value as Record<string, unknown>)[key] : undefined;
}

function countOf(value: unknown): number {
  return Number.isSafeInteger(value) && (value as number) >= 0 ? (value as number) : 0;
}

function countsOf(value: unknown): DraftCounts {
  return { kept: countOf(fieldOf(value, 'kept')), edited: countOf(fieldOf(value, 'edited')), dropped: countOf(fieldOf(value, 'dropped')) };
}

function parsed(stored: string | null): unknown {
  if (stored === null) return null;
  try {
    return JSON.parse(stored);
  } catch {
    return null;
  }
}

function tallyOf(stored: string | null): Record<AiModelName, DraftCounts> {
  const value = parsed(stored);
  return {
    'claude-haiku-5-5': countsOf(fieldOf(value, 'claude-haiku-5-5')),
    'claude-opus-5-5': countsOf(fieldOf(value, 'claude-opus-5-5')),
    rules: countsOf(fieldOf(value, 'rules')),
  };
}

export function addDraftOutcome(outcome: DraftOutcome, model: AiModelName, backend?: StoreBackend): Promise<void> {
  return serially(async () => {
    const store = await storeOf(backend);
    const tally = tallyOf(await store.getItem(DRAFT_TALLY_KEY));
    tally[model][outcome] += 1;
    await store.setItem(DRAFT_TALLY_KEY, JSON.stringify(tally));
  });
}

export function readDraftTally(backend?: StoreBackend): Promise<DraftTally> {
  return serially(async () => tallyOf(await (await storeOf(backend)).getItem(DRAFT_TALLY_KEY)));
}

export function resetDraftTally(backend?: StoreBackend): Promise<void> {
  return serially(async () => {
    await (await storeOf(backend)).removeItem(DRAFT_TALLY_KEY);
  });
}
```

```ts
import assert from 'node:assert/strict';
import { test } from 'node:test';

import type { StoreBackend } from '../wallet/store/walletStore.ts';
import { DRAFT_TALLY_KEY, addDraftOutcome, readDraftTally, resetDraftTally } from './draftTally.ts';

// P3-T22's export check looks for this exact key, so it is spelled out here.
const KEY = 'jotbun.dev.draft-tally';
const NONE = { kept: 0, edited: 0, dropped: 0 };
const ZEROS = { 'claude-haiku-5-5': NONE, 'claude-opus-5-5': NONE, rules: NONE };

function memory(seed: Record<string, string> = {}): StoreBackend & { rows: Map<string, string> } {
  const rows = new Map(Object.entries(seed));
  return {
    rows,
    getItem: async (key) => rows.get(key) ?? null,
    setItem: async (key, value) => { rows.set(key, value); },
    removeItem: async (key) => { rows.delete(key); },
    getAllKeys: async () => [...rows.keys()],
    multiRemove: async (keys) => { for (const key of keys) rows.delete(key); },
  };
}

test('the tally counts outcomes per model and holds nothing else', async () => {
  assert.equal(DRAFT_TALLY_KEY, KEY);
  const backend = memory();
  await addDraftOutcome('kept', 'claude-haiku-5-5', backend);
  await addDraftOutcome('kept', 'claude-haiku-5-5', backend);
  await addDraftOutcome('dropped', 'rules', backend);
  const expected = { 'claude-haiku-5-5': { kept: 2, edited: 0, dropped: 0 }, 'claude-opus-5-5': NONE, rules: { kept: 0, edited: 0, dropped: 1 } };
  assert.deepEqual(await readDraftTally(backend), expected);
  // One key, and its JSON holds only the three models, each with exactly those three counts.
  assert.deepEqual([...backend.rows.keys()], [KEY]);
  assert.deepEqual(JSON.parse(backend.rows.get(KEY) ?? 'null'), expected);
});

test('several outcomes added at once each count, and a read waits for the counts asked before it', async () => {
  const backend = memory();
  // As several drafts saved together (P3-T18) add them: none waits for the last.
  const adds = Array.from({ length: 5 }, () => addDraftOutcome('edited', 'claude-opus-5-5', backend));
  const read = readDraftTally(backend);
  assert.deepEqual((await read)['claude-opus-5-5'], { kept: 0, edited: 5, dropped: 0 });
  await Promise.all(adds);
  assert.deepEqual(JSON.parse(backend.rows.get(KEY) ?? 'null')['claude-opus-5-5'], { kept: 0, edited: 5, dropped: 0 });
});

test('a stored tally that is not the tally\'s shape reads as zeros, and the next count keeps counts only', async () => {
  for (const raw of ['{not json', 'null', '7', '"kept"', '[1,2]']) {
    assert.deepEqual(await readDraftTally(memory({ [KEY]: raw })), ZEROS, raw);
  }
  const backend = memory({

test('a count that cannot be stored rejects alone, and the next is counted', async () => {
  const backend = memory();
  let failures = 1;
  const flaky: StoreBackend = {
    ...backend,
    setItem: async (key, value) => {
      if (failures > 0) {
        failures -= 1;
        throw new Error('the disk is full');
      }
      await backend.setItem(key, value);
    },
  };
  await assert.rejects(addDraftOutcome('kept', 'rules', flaky), /the disk is full/);
  await assert.doesNotReject(addDraftOutcome('edited', 'rules', flaky));
  assert.deepEqual((await readDraftTally(flaky)).rules, { kept: 0, edited: 1, dropped: 0 });

test('reset removes the tally, and a later count starts again from zero', async () => {
  const backend = memory({ [KEY]: JSON.stringify({ 'claude-haiku-5-5': { kept: 4, edited: 1, dropped: 2 }, 'claude-opus-5-5': NONE, rules: NONE }) });

  assert.deepEqual(await readDraftTally(backend), ZEROS);

});
```
