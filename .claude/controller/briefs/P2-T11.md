```ts
// A wallet's look (W7, R64): its kind, its material and Bun's face on it. The
// person's choices are kept on the phone per account (WalletLooksProvider);
// a wallet with none kept gets a material and a face derived from its id, and
// no kind. Kept text is read leniently: a damaged entry is dropped, never
// thrown. Pure: a Node test reads it.
import { WALLET_MATERIALS } from '../../theme/walletPalette.ts';
import type { WalletMaterial } from '../../theme/walletPalette.ts';

/** §5.2's four (W25), in wc-wallet-type's order without the e-wallet: 1D-T1's `wallet_type` values. */
export type WalletKind = 'cash' | 'bank' | 'credit_card' | 'transit';
export const WALLET_KINDS: readonly WalletKind[] = ['cash', 'bank', 'credit_card', 'transit'];

/** ยิ้ม, ขยิบตา, สบาย ๆ, ดีใจ and ว้าว, in wc-wallet-edit's order. */
export type BunFaceName = 'smile' | 'wink' | 'calm' | 'happy' | 'wow';
export const BUN_FACES: readonly BunFaceName[] = ['smile', 'wink', 'calm', 'happy', 'wow'];

export type WalletLook = { kind: WalletKind | null; material: WalletMaterial; face: BunFaceName };

/** 32-bit FNV-1a over the text's UTF-16 code units, unsigned. */
function fnv1a(text: string): number {
  let hash = 0x811c9dc5;
  for (let index = 0; index < text.length; index += 1) {
    hash ^= text.charCodeAt(index);
    hash = Math.imul(hash, 0x01000193);
  }
  return hash >>> 0;
}

/** The look of a wallet with none kept: the same id always gets the same material and face, and no kind. */
export function derivedLook(walletId: string): WalletLook {
  const hash = fnv1a(walletId);
  return {
    kind: null,
    material: WALLET_MATERIALS[hash % WALLET_MATERIALS.length],
    face: BUN_FACES[(hash >>> 8) % BUN_FACES.length],
  };
}

/** Kept text as JSON; undefined when it is missing or is not JSON. */
function parsed(raw: string | null): unknown {
  if (raw === null) return undefined;
  try {
    return JSON.parse(raw);
  } catch {
    return undefined;
  }
}

/** A kept JSON object's own entries; none when the text is missing, is not JSON or is not an object. */
export function keptEntries(raw: string | null): [string, unknown][] {
  const value = parsed(raw);
  return value !== null && typeof value === 'object' && !Array.isArray(value) ? Object.entries(value) : [];
}

const isOneOf = <T extends string>(list: readonly T[], value: unknown): value is T => typeof value === 'string' && (list as readonly string[]).includes(value);

function asLook(value: unknown): WalletLook | null {
  if (value === null || typeof value !== 'object') return null;
  const { kind, material, face } = value as Record<string, unknown>;
  if (kind !== null && !isOneOf(WALLET_KINDS, kind)) return null;
  if (!isOneOf(WALLET_MATERIALS, material) || !isOneOf(BUN_FACES, face)) return null;
  return { kind, material, face };
}

/** The kept looks by wallet id (the account store's `looks`). A damaged entry is dropped, never thrown. */
export function parseLooks(raw: string | null): Readonly<Record<string, WalletLook>> {
  return Object.fromEntries(keptEntries(raw).flatMap(([walletId, value]) => {
    const look = asLook(value);
    return look === null ? [] : [[walletId, look] as const];
  }));
}

/** The kept wallet order (the account store's `wallet-order`): ids, each once, in order. Anything else is dropped. */
export function parseOrder(raw: string | null): readonly string[] {
  const value = parsed(raw);
  return Array.isArray(value) ? [...new Set(value.filter((id): id is string => typeof id === 'string'))] : [];
}
```

```ts
function boardShapes(markup: string): DrawingShape[] {
  return [...markup.matchAll(/<(path|circle|rect)\s([^>]*)>/g)].map(([, tag, attributes]) => {
    const attribute = (name: string) => new RegExp(`(?:^|\\s)${name}="([^"]*)"`).exec(attributes)?.[1] ?? '';
    const number = (name: string) => Number(attribute(name));
    if (tag === 'path') return { d: attribute('d') };
    if (tag === 'circle') {
      const circle = [number('cx'), number('cy'), number('r')] as const;
      return attribute('fill') === 'none' ? { circle } : { circle, filled: true };
    }
    return { rect: [number('x'), number('y'), number('width'), number('height')] as const, rx: number('rx') };
  });
}
```

`apps/app/src/wallet/looks/WalletLooksProvider.tsx`:

```tsx
// Looks kept on the phone (W7, R64): each wallet's kind, material and Bun
// face, the wallet order and each category's icon. They are the person's
// data, kept per account in the account store (P2-T3) under `looks`,
// `wallet-order` and `category-icons` for the open diary's account, so they
// survive sign-out and go only with a confirmed account deletion. With no
// account (no diary open: the web, opening, locked or signed out) every
// wallet shows its derived look and the setters keep nothing. WalletShell
// mounts the one provider (P2-T18); outside it, looks are derived, no order
// or icon is kept, and nothing changes.
//
// Each change reads its name from the store, changes it and writes it back,
// after the account's first read and after the change before it: two quick
// changes never drop each other, and a change made just before the account
// changes is kept for the account it was made in. A name the store cannot
// read counts as nothing kept and is never written over; a change to it, or a
// write that fails, shows until the app closes.
import type { CategoryDto } from '@gogospend/contracts';
import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from 'react';
import type { ReactNode } from 'react';

import { useDiaryView } from '@mobile/spend/v1/SpendDiaryContext';
import { walletAccountStore } from '../store/deviceWalletStore';
import type { KeyValueStore } from '../store/walletStore';
import type { GlyphKey } from './categoryGlyphs';
import { iconFor, parseIcons } from './categoryIcons';
import type { KeptIcons } from './categoryIcons';
import { derivedLook, parseLooks, parseOrder } from './walletLooks';
import type { WalletLook } from './walletLooks';

/** The account store's names: `jotbun.account.<accountId>.looks`, `.wallet-order` and `.category-icons`. */
export const LOOKS_NAME = 'looks';
export const WALLET_ORDER_NAME = 'wallet-order';
export const CATEGORY_ICONS_NAME = 'category-icons';

type KeptLooks = Readonly<Record<string, WalletLook>>;

type Kept = {
  /** The account these were read for. */
  readonly accountId: string | null;
  readonly looks: KeptLooks;
  readonly order: readonly string[];
  readonly icons: KeptIcons;
};

/** One kept name: how its text reads, and where its value sits. */
type Slot<T> = {
  readonly name: string;
  readonly parse: (raw: string | null) => T;
  readonly of: (kept: Kept) => T;
  readonly put: (kept: Kept, value: T) => Kept;
};

const LOOKS: Slot<KeptLooks> = { name: LOOKS_NAME, parse: parseLooks, of: (kept) => kept.looks, put: (kept, looks) => ({ ...kept, looks }) };
const ORDER: Slot<readonly string[]> = { name: WALLET_ORDER_NAME, parse: parseOrder, of: (kept) => kept.order, put: (kept, order) => ({ ...kept, order }) };
const ICONS: Slot<KeptIcons> = { name: CATEGORY_ICONS_NAME, parse: parseIcons, of: (kept) => kept.icons, put: (kept, icons) => ({ ...kept, icons }) };

const nothingKept = (accountId: string | null): Kept => ({ accountId, looks: {}, order: [], icons: {} });

export type WalletLooks = {
  /** The kept look, else the one derived from the id. */
  lookOf(walletId: string): WalletLook;
  /** Keeps a wallet's look for the open account; with none open, keeps nothing. */
  setLook(walletId: string, look: WalletLook): Promise<void>;
  /** The kept wallet order, by id; empty when none is kept. */
  order: readonly string[];
  /** Keeps the wallet order (each id once) for the open account; with none open, keeps nothing. */
  setOrder(ids: readonly string[]): Promise<void>;
};

export type CategoryIcons = {
  /** The kept icon, else the pack's glyph, else the built-in key, else null (the initial). */
  iconOf(category: Pick<CategoryDto, 'id' | 'type' | 'label' | 'builtin_key'>): GlyphKey | null;
  /** Keeps a category's icon for the open account; with none open, keeps nothing. */
  setIcon(categoryId: string, icon: GlyphKey): Promise<void>;
};

type LooksContextValue = {
  readonly kept: Kept;
  readonly setLook: WalletLooks['setLook'];
  readonly setOrder: WalletLooks['setOrder'];
  readonly setIcon: CategoryIcons['setIcon'];
};

const keepNothing = () => Promise.resolve();

/** Outside the provider: derived looks, no order, no kept icons, and the setters keep nothing. */
const OUTSIDE: LooksContextValue = { kept: nothingKept(null), setLook: keepNothing, setOrder: keepNothing, setIcon: keepNothing };

const LooksContext = createContext<LooksContextValue>(OUTSIDE);

/** A name's text (null when absent), or null in place of the whole answer when the store cannot read it. */
const readName = (store: KeyValueStore, name: string) => store.get(name).then((raw) => ({ raw }), () => null);

export function WalletLooksProvider({ children }: { children: ReactNode }) {
  const { accountId } = useDiaryView();
  const [state, setState] = useState<Kept>(() => nothingKept(accountId));
  // Every read and every change runs after the one before it.
  const queue = useRef<Promise<void>>(Promise.resolve());
  const enqueue = useCallback((step: () => Promise<void>) => {
    const run = queue.current.then(step);
    queue.current = run.catch(() => undefined);
    return run;
  }, []);

  useEffect(() => {
    setState((now) => (now.accountId === accountId ? now : nothingKept(accountId)));
    if (accountId === null) return;
    const store = walletAccountStore(accountId);
    void enqueue(async () => {
      const [looks, order, icons] = await Promise.all([LOOKS, ORDER, ICONS].map((slot) => readName(store, slot.name)));
      const read: Kept = { accountId, looks: parseLooks(looks?.raw ?? null), order: parseOrder(order?.raw ?? null), icons: parseIcons(icons?.raw ?? null) };
      setState((now) => (now.accountId === accountId ? read : now));
    });
  }, [accountId, enqueue]);

  const keep = useCallback(<T,>(slot: Slot<T>, change: (value: T) => T): Promise<void> => {
    if (accountId === null) return Promise.resolve();
    const store = walletAccountStore(accountId);
    return enqueue(async () => {
      const stored = await readName(store, slot.name);
      const value = stored === null ? null : change(slot.parse(stored.raw));
      setState((now) => (now.accountId === accountId ? slot.put(now, value ?? change(slot.of(now))) : now));
      if (value !== null) await store.set(slot.name, JSON.stringify(value)).catch(() => undefined);
    });
  }, [accountId, enqueue]);

  const setLook = useCallback((walletId: string, look: WalletLook) => keep(LOOKS, (looks) => ({ ...looks, [walletId]: look })), [keep]);
  const setOrder = useCallback((ids: readonly string[]) => keep(ORDER, () => [...new Set(ids)]), [keep]);
  const setIcon = useCallback((categoryId: string, icon: GlyphKey) => keep(ICONS, (icons) => ({ ...icons, [categoryId]: icon })), [keep]);

  // Right after the account changes, nothing of the last account's shows, even for one render.
  const kept = useMemo(() => (state.accountId === accountId ? state : nothingKept(accountId)), [state, accountId]);
  const value = useMemo<LooksContextValue>(() => ({ kept, setLook, setOrder, setIcon }), [kept, setLook, setOrder, setIcon]);
  return <LooksContext.Provider value={value}>{children}</LooksContext.Provider>;
}

/** Wallet looks and order as the nearest provider keeps them; outside one, derived looks and no order. */
export function useWalletLooks(): WalletLooks {
  const { kept, setLook, setOrder } = useContext(LooksContext);
  return useMemo(() => ({
    lookOf: (walletId: string) => kept.looks[walletId] ?? derivedLook(walletId),
    setLook,
    order: kept.order,
    setOrder,
  }), [kept, setLook, setOrder]);
}

/** Category icons as the nearest provider keeps them; outside one, none are kept. */
export function useCategoryIcons(): CategoryIcons {
  const { kept, setIcon } = useContext(LooksContext);
  const { icons } = kept;
  return useMemo(() => ({ iconOf: (category) => iconFor(category, icons), setIcon }), [icons, setIcon]);
}
```

```tsx
/**
 * A store in memory whose first read of each name, until `release()`, waits and then answers what
 * the name held when it was asked, as a slow read in flight does. Every other read answers at once.
 */
function slowFirstReads(seed: Readonly<Record<string, string>>) {
  const stored = new Map(Object.entries(seed));
  const held: (() => void)[] = [];
  const asked = new Set<string>();
  let open = false;
  const backend: StoreBackend = {
    getItem: (key) => {
      const value = stored.get(key) ?? null;
      if (open || asked.has(key)) return Promise.resolve(value);
      asked.add(key);
      return new Promise((resolve) => held.push(() => resolve(value)));
    },
    setItem: async (key, value) => { stored.set(key, value); },
    removeItem: async (key) => { stored.delete(key); },
    getAllKeys: async () => [...stored.keys()],
    multiRemove: async (keys) => { for (const key of keys) stored.delete(key); },
  };
  return {
    stored,
    backend,
    release() {
      open = true;
      for (const answer of held.splice(0)) answer();
    },
  };
}

/** A store in memory whose every read waits until the next `release()`, then answers what the name held when it was asked. */
function heldReads(seed: Readonly<Record<string, string>>) {
  const stored = new Map(Object.entries(seed));
  const held: (() => void)[] = [];
  const backend: StoreBackend = {
    getItem: (key) => {
      const value = stored.get(key) ?? null;
      return new Promise((resolve) => held.push(() => resolve(value)));
    },
    setItem: async (key, value) => { stored.set(key, value); },
    removeItem: async (key) => { stored.delete(key); },
    getAllKeys: async () => [...stored.keys()],
    multiRemove: async (keys) => { for (const key of keys) stored.delete(key); },
  };
  return {
    backend,
    /** Answers every read asked so far. */
    release() {
      for (const answer of held.splice(0)) answer();
    },
  };
}

/**
 * One button group of a board, parsed on its own: the whole page would load
 * its web fonts and script, and the tests never reach the network.
 */
function boardGroup(file: string, labelledBy: string): DocumentFragment {
  const html = readFileSync(join(BOARDS, file), 'utf8');
  const at = html.indexOf(`aria-labelledby="${labelledBy}"`);
  const template = document.createElement('template');
  template.innerHTML = html.slice(html.lastIndexOf('<div', at), html.indexOf('</div>', at) + '</div>'.length);
  return template.content;
}

/** A board button's SVG shapes, by the attributes that draw them; an absent fill is none. */
function shapesOf(svg: Element) {
  return [...svg.children].map((node) => {
    const value = (name: string) => node.getAttribute(name);
    const number = (name: string) => (value(name) === null ? null : Number(value(name)));
    return {
      tag: node.tagName.toLowerCase(),
      d: value('d'),
      at: [number('cx'), number('cy'), number('r'), number('rx'), number('ry')],
      fill: value('fill') ?? 'none',
      stroke: value('stroke'),
      width: number('stroke-width'),
      cap: value('stroke-linecap'),
    };
  });
}
```

```tsx
const KIND_LINE: Readonly<Record<WalletKind, keyof typeof COPY>> = {
  cash: 'card.kindCash',
  bank: 'card.kindBank',
  credit_card: 'card.kindCard',
  transit: 'card.kindTransit',
};

const CARD_COPY = { ...COMMON_COPY, ...COPY };

/** The size of a canonical satang string. */
const size = (minor: Minor): Minor => (minor.startsWith('-') ? minor.slice(1) : minor);

/** A pending total or today's money out worth a line: none, null and zero draw nothing. */
const worthALine = (minor: Minor | null | undefined): Minor | null => (minor === undefined || minor === null || minor === '0' ? null : minor);

/** The card's words. `t` reads COMMON_COPY and the card's COPY. */
export function walletCardWords(wallet: WalletCardModel, look: WalletLook, hidden: boolean, t: WalletT): WalletCardWords {
  // A credit card reads what was spent instead of a balance (W25, wc-wallet-type), from zero down.
  const spending = look.kind === 'credit_card' && (wallet.balanceMinor === '0' || wallet.balanceMinor.startsWith('-'));
  const amountMinor = spending ? size(wallet.balanceMinor) : wallet.balanceMinor;
  const pending = worthALine(wallet.pendingMinor);
  const today = worthALine(wallet.todayOutMinor);
  // Each line keeps its place in private mode, so nothing moves (W10); only its amount is masked.
  const pendingLine = pending === null ? null : t('card.pending', { amount: hidden ? HIDDEN_AMOUNT : walletMoneyText(pending) });
  const todayLine = today === null ? null : t('card.todayOut', { amount: hidden ? HIDDEN_AMOUNT : walletMoneyText(today, { sign: 'expense' }) });
  const said = hidden
    ? [wallet.name, t('common.amountHidden')]
    : [
        wallet.name,
        t(spending ? 'card.spokenSpent' : 'card.spokenLeft', { amount: spokenMoney(amountMinor, t) }),
        ...(pending === null ? [] : [t('card.spokenPending', { amount: spokenMoney(size(pending), t) })]),
        ...(today === null ? [] : [t('card.spokenToday', { amount: spokenMoney(size(today), t) })]),
      ];
  return {
    amountMinor,
    amountPrefix: spending ? t('card.spent') : null,
    kindLine: look.kind === null ? null : t(KIND_LINE[look.kind]),
    pendingLine,
    todayLine,
    spoken: said.join(', '),
  };
}
```
