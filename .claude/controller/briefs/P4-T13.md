// installed, so the app reads no album: pickerPhotoLibrary() picks slips by hand through the
// installed expo-image-picker 58.0.10, and each picked photo is the picker's own copy in the
// app's cache, which release() deletes once the scan engine (P4-T14) is done with it. Q1's
// mediaLibraryPhotoLibrary() waits on the owner's yes and is not written here.
// expo-image-picker, expo-file-system and expo-linking load through optional loaders

import { SLIP_LIMITS } from '@gogospend/contracts';

export type PhotoAccess = 'all' | 'limited' | 'none' | 'undetermined';
export type AlbumInfo = { id: string; title: string; count: number };
/**
 * A photo the scan can read: an album photo, or a picked photo's copy (albumId null; its id is the
 * picker's assetId, else its uri). bytes is the size the phone reports, a hint only: Android's
 * picker gives the original photo's size, not its copy's, so the shrink port measures the file.

export type PhotoLibraryPort = {
  readonly albumsSupported: boolean;
  access(): Promise<PhotoAccess>;
  request(): Promise<PhotoAccess>;
  albums(): Promise<AlbumInfo[]>;
  /** An album's photos, oldest first, strictly after (ms, id); a null id comes before every id, so it takes the photos at ms too. */
  assetsSince(albumId: string, after: { ms: number; id: string | null }, limit: number): Promise<PhotoAsset[]>;
  /** An album photo again by its id (P4-T14's resume); null once it is gone. */
  asset(id: string): Promise<PhotoAsset | null>;
  pick(limit: number): Promise<PhotoAsset[]>;
  /** Deletes a picked photo's copy, best effort; nothing for an album photo. */

  /** Opens the app's page in the phone's settings, best effort. */
  openSettings(): Promise<void>;

type FileSystemModule = typeof import('expo-file-system');
type LinkingModule = typeof import('expo-linking');

export type PickerPhotoLibraryOptions = {
  loadModule?: () => Promise<ImagePickerModule>;
  loadFileSystem?: () => Promise<FileSystemModule>;
  loadLinking?: () => Promise<LinkingModule>;
};

// The picker's media library answer: granted is all or limited; undetermined until asked; else none.
function photoAccessOf(answer: { status: string; accessPrivileges?: 'all' | 'limited' | 'none' }): PhotoAccess {
  if (answer.status === 'granted') return answer.accessPrivileges === 'limited' ? 'limited' : 'all';
  return answer.status === 'undetermined' ? 'undetermined' : 'none';
}

/** Q1's default: no albums and no lookup by id (P4-T14 finds a picked copy by its uri); pick() multi-selects at quality 0.7. */
export function pickerPhotoLibrary(options: PickerPhotoLibraryOptions = {}): PhotoLibraryPort {
  const loadPicker = options.loadModule ?? (() => import('expo-image-picker'));
  const loadFileSystem = options.loadFileSystem ?? (() => import('expo-file-system'));
  const loadLinking = options.loadLinking ?? (() => import('expo-linking'));
  return {
    albumsSupported: false,
    access: async () => photoAccessOf(await (await loadPicker()).getMediaLibraryPermissionsAsync()),
    request: async () => photoAccessOf(await (await loadPicker()).requestMediaLibraryPermissionsAsync()),
    albums: async () => [],
    assetsSince: async () => [],
    asset: async () => null,
    async pick(limit) {
      const picker = await loadPicker();
      const result = await picker.launchImageLibraryAsync({
        mediaTypes: ['images'],
        allowsMultipleSelection: true,
        selectionLimit: limit,
        quality: SLIP_LIMITS.jpegQuality,
        preferredAssetRepresentationMode: picker.UIImagePickerPreferredAssetRepresentationMode.Compatible,
      });
      if (result.canceled) return [];
      const createdMs = Date.now(); // the picker gives no taken time: a pick is dated when it is made
      return result.assets.map((asset) => ({
        id: asset.assetId ?? asset.uri,
        uri: asset.uri,
        createdMs,
        albumId: null,
        ...(typeof asset.fileSize === 'number' ? { bytes: asset.fileSize } : {}),
      }));
    },
    async release(asset) {
      if (asset.albumId !== null) return; // an album photo is the person's own
      try {
        const { File } = await loadFileSystem();
        new File(asset.uri).delete();
      } catch {
        // Best effort: a copy already gone, or one the phone will not delete, stays in the cache the phone clears.
      }
    },
    async openSettings() {
      try {
        await (await loadLinking()).openSettings();
      } catch {
        // Best effort: where settings cannot open (the web), nothing happens.
      }
    },
  };
}

// Making a slip photo small enough to send (P4-T13). Under owner Q2's default
// expo-image-manipulator is not installed, so pickerShrink() sends a photo's own bytes when they
// fit SLIP_LIMITS.imageBytes (1,200,000; a picked JPEG is already compressed at quality 0.7) and
// otherwise nothing: a larger photo, such as a PNG screenshot iOS will not compress, stays
// unread on the phone. Q2's manipulatorShrink() waits on the owner's yes and is not written here.
// expo-file-system 58.0.9 loads through an optional loader (src/lib/haptics.ts's pattern), so
// Node loads this module.
import { SLIP_LIMITS } from '@gogospend/contracts';

import type { PhotoAsset } from './photoLibrary.ts';

export type ShrunkImage = { uri: string | null; base64: string; bytes: number; mediaType: 'image/jpeg' | 'image/png' | 'image/webp' };
/** shrink(): null when the photo cannot be sent (too large, or not a JPEG, PNG or WebP); it throws when the photo is gone. */
export type ImageShrinkPort = { shrink(asset: PhotoAsset): Promise<ShrunkImage | null>; remove(image: ShrunkImage): Promise<void> };

type FileSystemModule = typeof import('expo-file-system');

// The contract's image text (parseSlipImage in packages/contracts/src/v1/slips.ts): a batch with
// anything else in it is refused whole.
const BASE64 = /^[A-Za-z0-9+/]+={0,2}$/;

function decodedBytes(base64: string): number {
  return (base64.length / 4) * 3 - (base64.endsWith('==') ? 2 : base64.endsWith('=') ? 1 : 0);
}

// The type an image's first bytes name.
function imageTypeOf(base64: string): ShrunkImage['mediaType'] | null {
  const head = atob(base64.slice(0, 16));
  if (head.startsWith('\xff\xd8\xff')) return 'image/jpeg';
  if (head.startsWith('\x89PNG\r\n\x1a\n')) return 'image/png';
  if (head.startsWith('RIFF') && head.slice(8, 12) === 'WEBP') return 'image/webp';
  return null;
}

/**
 * Q2's default: a QR-checked photo's or a shared file's bytes as they are, within 1,200,000
 * bytes, else null. A photo whose size says it is too large is never read. A copy that is gone
 * throws (its read fails), so the scan engine keeps it apart from an unreadable photo (P4-T14).
 * The image's uri is null: nothing was copied, so remove() deletes nothing, never a pending copy.
 */
export function pickerShrink(options: { loadModule?: () => Promise<FileSystemModule> } = {}): ImageShrinkPort {
  const load = options.loadModule ?? (() => import('expo-file-system'));
  return {
    async shrink(asset) {
      const { File } = await load();
      const file = new File(asset.uri);
      if (file.size > SLIP_LIMITS.imageBytes) return null;
      const base64 = await file.base64();
      if (base64.length % 4 !== 0 || !BASE64.test(base64)) return null;
      const bytes = decodedBytes(base64); // the bytes read, also where the phone could not give a size (0)
      if (bytes > SLIP_LIMITS.imageBytes) return null;
      const mediaType = imageTypeOf(base64);
      return mediaType === null ? null : { uri: null, base64, bytes, mediaType };
    },
    async remove() {
      // Nothing to delete: the image is the photo's own bytes.
    },
  };
}

function albumLabel(album: string | null): string | null {
  if (album === null) return null;
  const label = [...album.normalize('NFC').trim()].slice(0, ALBUM_MAX).join('').trim();
  return label === '' ? null : label;
}

function slipBatchItem(slip: ReadySlip): SlipBatchItem {
  return { ref_hash: slip.refHash, image_base64: slip.image.base64, media_type: slip.image.mediaType, album: albumLabel(slip.album), origin: slip.origin };
}

/** The POST /v1/slips body for these slips. */
export function slipBatchRequest(slips: readonly ReadySlip[], header: Omit<SlipBatchRequest, 'slips'>): SlipBatchRequest {
  return { ...header, slips: slips.map(slipBatchItem) };
}

/** The UTF-8 length of the JSON body the transport sends for these slips. */
export function slipBatchBytes(slips: readonly ReadySlip[], header: Omit<SlipBatchRequest, 'slips'>): number {
  return utf8Length(JSON.stringify(slipBatchRequest(slips, header)));
}

/**
 * The slips in order, in batches of at most 5 slips and 8 MiB of JSON. Each batch is measured
 * with `last: false`, the longer form, so it fits whichever `last` it is sent with. A slip too
 * large for a batch of its own throws RangeError (a shrink port never makes one).
 */
export function buildSlipBatches(slips: readonly ReadySlip[], header: Omit<SlipBatchRequest, 'slips'>): ReadySlip[][] {
  const empty = slipBatchBytes([], { ...header, last: false });
  const batches: ReadySlip[][] = [];
  let batch: ReadySlip[] = [];
  let bytes = empty;
  for (const slip of slips) {
    const size = utf8Length(JSON.stringify(slipBatchItem(slip)));
    if (empty + size > SLIP_LIMITS.batchBytes) throw new RangeError('A slip is larger than a batch.');
    if (batch.length === SLIP_LIMITS.batchSlips || (batch.length > 0 && bytes + 1 + size > SLIP_LIMITS.batchBytes)) {
      batches.push(batch);
      batch = [];
      bytes = empty;
    }
    bytes += batch.length === 0 ? size : 1 + size; // a comma before every slip but the first
    batch.push(slip);
  }
  if (batch.length > 0) batches.push(batch);
  return batches;
}

export type FakePhotoLibrary = PhotoLibraryPort & { released: PhotoAsset[]; calls: FakeLibraryCall[] };
export type FakeImageShrink = ImageShrinkPort & { removed: ShrunkImage[] };

// Oldest first, ties broken by id (code unit order, as SQLite's BINARY collation).
const byAge = (a: PhotoAsset, b: PhotoAsset): number => a.createdMs - b.createdMs || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0);

/**
 * A photo library over the given albums and picks. access() and request() both answer `access`
 * (default 'all'), albumsSupported defaults to true, pick(limit) gives the first `limit` of
 * `picked`, and asset(id) finds a listed album photo only by its id (never by its uri, never a
 * pick). It records every call in `calls` and every released photo in `released`.
 */
export function fakePhotoLibrary(
  options: { access?: PhotoAccess; albums?: readonly FakeAlbum[]; picked?: readonly PhotoAsset[]; albumsSupported?: boolean } = {},
): FakePhotoLibrary {
  const access = options.access ?? 'all';
  const albums = (options.albums ?? []).map((album) => ({
    id: album.id,
    title: album.title,
    assets: (album.assets ?? []).map((asset): PhotoAsset => ({ ...asset, albumId: album.id })).sort(byAge),
  }));
  const released: PhotoAsset[] = [];
  const calls: FakeLibraryCall[] = [];
  const record = (method: FakeLibraryCall['method'], ...args: unknown[]): void => {
    calls.push({ method, args });
  };
  return {
    albumsSupported: options.albumsSupported ?? true,
    released,
    calls,
    async access() {
      record('access');
      return access;
    },
    async request() {
      record('request');
      return access;
    },
    async albums() {
      record('albums');
      return albums.map(({ id, title, assets }) => ({ id, title, count: assets.length }));
    },
    async assetsSince(albumId, after, limit) {
      record('assetsSince', albumId, { ...after }, limit);
      const album = albums.find((candidate) => candidate.id === albumId);
      return (album?.assets ?? [])
        .filter((asset) => asset.createdMs > after.ms || (asset.createdMs === after.ms && (after.id === null || asset.id > after.id)))
        .slice(0, limit)
        .map((asset) => ({ ...asset }));
    },
    async asset(id) {
      record('asset', id);
      for (const album of albums) {
        const found = album.assets.find((asset) => asset.id === id);
        if (found) return { ...found };
      }
      return null;
    },
    async pick(limit) {
      record('pick', limit);
      return (options.picked ?? []).slice(0, limit).map((asset) => ({ ...asset }));
    },
    async release(asset) {
      record('release', asset);
      released.push(asset);
    },
    async openSettings() {
      record('openSettings');
    },
  };
}

/** The fake shrink's image of a photo: a JPEG's first bytes, then the photo's uri, so a test can tell which copy was sent. */
export function fakeShrunkImage(asset: PhotoAsset): ShrunkImage {
  const bytes = `\xff\xd8\xff${encodeURIComponent(asset.uri)}`;
  return { uri: null, base64: btoa(bytes), bytes: bytes.length, mediaType: 'image/jpeg' };
}

/**
 * A shrink port: fakeShrunkImage for each photo, null for a photo whose id is in `tooLarge`, and a
 * throw for a copy whose uri is in `gone` (a removed copy, P4-T14). It records each image removed.
 */
export function fakeImageShrink(options: { tooLarge?: readonly string[]; gone?: readonly string[] } = {}): FakeImageShrink {

    removed,
    async shrink(asset) {

      return options.tooLarge?.includes(asset.id) ? null : fakeShrunkImage(asset);
    },
    async remove(image) {
      removed.push(image);
    },
  };
}

 * reads it and throws for a missing one (iOS's Data(contentsOf:), Android's stream), and each read

 */
function fakeFiles(files: Record<string, string>, sizes: Record<string, number> = {}) {
  const held = new Map(Object.entries(files));
  const calls: string[] = [];

    readonly uri: string;
    constructor(uri: string) {
      this.uri = uri;

      const text = held.get(this.uri);

      calls.push(`delete ${this.uri}`);
      if (!held.delete(this.uri)) throw new Error('no such file');

// P4-T13: slip batches for POST /v1/slips: at most 5 slips and 8 MiB of JSON each, in order.
import assert from 'node:assert/strict';
import { Buffer } from 'node:buffer';
import { test } from 'node:test';

import { SLIP_LIMITS, parseSlipBatchRequest } from '@gogospend/contracts';
import type { SlipBatchRequest } from '@gogospend/contracts';

import { buildSlipBatches, slipBatchBytes, slipBatchRequest } from './slipBatches.ts';
import type { ReadySlip } from './slipBatches.ts';

type Header = Omit<SlipBatchRequest, 'slips'>;

const MIB = 1024 * 1024;
const LIMIT = 8 * MIB;
const HEADER: Header = { purpose: 'backfill', last: false, today: '2026-11-09', locale: 'th' };

const slip = (n: number, base64: string, album: string | null = null): ReadySlip => ({
  assetId: `photo-${n}`,
  refHash: n.toString(16).padStart(64, '0'),
  image: { uri: null, base64, bytes: (base64.length / 4) * 3, mediaType: 'image/jpeg' },
  album,
  origin: 'album',
});
/** The wire body, built here apart from the module, and its UTF-8 length. */
const body = (slips: readonly ReadySlip[], header: Header) => ({
  ...header,
  slips: slips.map((item) => ({ ref_hash: item.refHash, image_base64: item.image.base64, media_type: item.image.mediaType, album: item.album, origin: item.origin })),
});
const bodyBytes = (slips: readonly ReadySlip[], header: Header): number => Buffer.byteLength(JSON.stringify(body(slips, header)), 'utf8');
const ids = (batches: readonly ReadySlip[][]): string[][] => batches.map((batch) => batch.map((item) => item.assetId));

test('batches hold at most 5 slips and 8 MiB of JSON, in order', () => {
  assert.equal(SLIP_LIMITS.batchBytes, LIMIT);
  const small = [1, 2, 3, 4, 5, 6].map((n) => slip(n, 'QUJD'));
  assert.deepEqual(ids(buildSlipBatches(small, HEADER)), [['photo-1', 'photo-2', 'photo-3', 'photo-4', 'photo-5'], ['photo-6']]);
  const threeMib = 'A'.repeat(3 * MIB);
  assert.deepEqual(ids(buildSlipBatches([1, 2, 3].map((n) => slip(n, threeMib)), HEADER)), [['photo-1', 'photo-2'], ['photo-3']]);
  const four = buildSlipBatches([1, 2, 3, 4].map((n) => slip(n, threeMib)), HEADER);
  assert.deepEqual(ids(four), [['photo-1', 'photo-2'], ['photo-3', 'photo-4']]);
  for (const batch of four) assert.ok(bodyBytes(batch, HEADER) <= LIMIT);
  assert.deepEqual(buildSlipBatches([], HEADER), []);
});

test("the byte count is the JSON body's UTF-8 length", () => {
  const header: Header = { ...HEADER, last: true };
  const slips = [slip(1, 'QUJD', 'กสิกรไทย'), slip(2, '/9j/4AAQ')];
  const expected = body(slips, header);
  const text = JSON.stringify(expected);
  assert.deepEqual(slipBatchRequest(slips, header), expected);
  assert.equal(slipBatchBytes(slips, header), Buffer.byteLength(text, 'utf8'));
  assert.equal(Buffer.byteLength(text, 'utf8'), text.length + 16); // each of the album's 8 Thai characters is 3 bytes
  assert.equal(slipBatchBytes([], header), Buffer.byteLength(JSON.stringify({ ...header, slips: [] }), 'utf8'));
  assert.deepEqual(parseSlipBatchRequest(JSON.parse(text)), expected); // the server's parser takes the body as it is
});

test('a batch is cut at exactly 8 MiB, counting the commas between slips, and is measured with last false', () => {
  const empty = bodyBytes([], HEADER);
  const overhead = bodyBytes([slip(1, '')], HEADER) - empty; // one slip's JSON, its image aside
  const twoMib = 'A'.repeat(2 * MIB);
  // Three slips whose body is exactly 8 MiB, two commas included, and then one byte more.
  const three = (extra: number) => [slip(1, twoMib), slip(2, twoMib), slip(3, 'A'.repeat(LIMIT - empty - 3 * overhead - 2 - 2 * twoMib.length + extra))];
  assert.equal(bodyBytes(three(0), HEADER), LIMIT);
  assert.equal(slipBatchBytes(three(0), HEADER), LIMIT);
  assert.deepEqual(ids(buildSlipBatches(three(0), HEADER)), [['photo-1', 'photo-2', 'photo-3']]);
  assert.deepEqual(ids(buildSlipBatches(three(1), HEADER)), [['photo-1', 'photo-2'], ['photo-3']]);
  // Sent with last: true, those would be exactly 8 MiB, but a batch is measured with last: false.
  const last: Header = { ...HEADER, last: true };
  assert.equal(bodyBytes(three(1), last), LIMIT);
  assert.deepEqual(ids(buildSlipBatches(three(1), last)), [['photo-1', 'photo-2'], ['photo-3']]);
});

test('a slip that fills a batch alone is sent alone, and one larger than a batch throws', () => {
  const empty = bodyBytes([], HEADER);
  const overhead = bodyBytes([slip(1, '')], HEADER) - empty;
  const alone = (extra: number) => slip(1, 'A'.repeat(LIMIT - empty - overhead + extra));
  assert.equal(bodyBytes([alone(0)], HEADER), LIMIT);
  assert.deepEqual(ids(buildSlipBatches([alone(0), slip(2, 'QUJD')], HEADER)), [['photo-1'], ['photo-2']]);
  assert.throws(() => buildSlipBatches([slip(2, 'QUJD'), alone(1)], HEADER), RangeError);
});

test("each slip's album is sent as the server keeps it: NFC, trimmed, at most 60 characters, and null when nothing is left", () => {
  const thai = 'ก'.repeat(58);
  const cutOnSpace = `  ${'ข'.repeat(59)} ค`; // trimmed, its 60th character is a space
  const albums = ['  K PLUS  ', 'Cafe\u0301', `${thai}\u{1D400}\u{1D401}\u{1D402}`, cutOnSpace, '   '];
  const slips = albums.map((album, index) => slip(index + 1, 'QUJD', album));
  const sent = slipBatchRequest(slips, HEADER).slips.map((item) => item.album);
  assert.deepEqual(sent, ['K PLUS', 'Caf\u00e9', `${thai}\u{1D400}\u{1D401}`, 'ข'.repeat(59), null]);
  assert.deepEqual(parseSlipBatchRequest(slipBatchRequest(slips, HEADER)).slips.map((item) => item.album), sent); // the server keeps what was sent
});
