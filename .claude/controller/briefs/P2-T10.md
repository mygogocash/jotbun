```ts
// Money text for the Bun Wallet (W9, §5.1): ฿ first, with thousands commas,
// over formatMinor, so satang never pass through a JS number. The compact form
// drops .00 only when the satang are zero; the exact form always shows two
// decimals. An expense takes the true minus (U+2212), income a plus, a balance
// a minus only below zero, and zero is never signed. Pure: a Node test reads it.
import type { Minor } from '@gogospend/contracts';

import type { WalletT } from '../copy/draftText.ts';
import { formatMinor } from '../../spend/v1/money.ts';

/**
 * `none`: the size of the amount, as a list row draws it (the row says what
 * it is). `expense` and `income`: that size after − or +. `balance`: a minus
 * only below zero.
 */
export type MoneySign = 'none' | 'expense' | 'income' | 'balance';

/** Every amount in private mode (W10, D41), drawn at the amount's own size. */
export const HIDDEN_AMOUNT = '฿••••';

// U+2212, the true minus sign: as wide as the plus, as the Bun diary's Amount draws it.
const MINUS = '\u2212';

/** `฿12,480`, `฿50.25`, `−฿50`, `+฿35,000`. Anything but a canonical satang string throws a TypeError (formatMinor's). */
export function walletMoneyText(minor: Minor, options: { sign?: MoneySign; exact?: boolean } = {}): string {
  const { sign = 'none', exact = false } = options;
  const text = formatMinor(minor, 'th');
  const negative = text.startsWith('-');
  const size = negative ? text.slice(1) : text;
  const shown = !exact && size.endsWith('.00') ? size.slice(0, -3) : size;
  if (minor === '0') return shown;
  if (sign === 'expense') return `${MINUS}${shown}`;
  if (sign === 'income') return `+${shown}`;
  if (sign === 'balance' && negative) return `${MINUS}${shown}`;
  return shown;
}

/**
 * A per-day figure in whole baht, rounded down (W9, C4): 690.99 a day reads
 * ฿690. It is never negative: budgetPace has no per-day figure once over.
 */
export function perDayText(perDayMinor: Minor): string {
  return walletMoneyText(perDayMinor, { exact: true }).slice(0, -3);
}

/**
 * What a screen reader says (§4: ฿ is read as บาท): `12,480 บาท`; with
 * satang, `50 บาท 25 สตางค์`; below zero, `ติดลบ 3,920 บาท`
 * (`common.baht`, `common.satang`, `common.minus`).
 */
export function spokenMoney(minor: Minor, t: WalletT): string {
  const text = formatMinor(minor, 'th');
  const negative = text.startsWith('-');
  const [baht, satang] = text.slice(negative ? 2 : 1).split('.');
  const amount = satang === '00'
    ? t('common.baht', { amount: baht })
    : t('common.satang', { baht, satang: satang.replace(/^0/, '') });
  return negative ? t('common.minus', { amount }) : amount;
}
```

```tsx
// A Bun Wallet amount (§5.1, W9, W10, W30): DM Sans 700 in every language,
// ฿ first with thousands commas, compact unless `exact`. It grows with the
// text size up to 64 pt and never more than body text, and is never clamped
// to a line. Income is violet unless a tone or a colour says otherwise. Its
// label is `label`, then the amount in words (฿ read as บาท). In private mode
// it is ฿•••• at its own size and line, so rows never change height, and it
// says only that the amount is hidden, never the dots.
import type { Minor } from '@gogospend/contracts';

import { Text } from '@mobile/theme/AppText';
import { useWalletColors } from '@mobile/theme/ThemeProvider';
import { LATIN_UI_FONT_FAMILY } from '@mobile/theme/uiFontFamily';
import { amountMaxScale, walletLineHeight } from '@mobile/theme/walletType';
import { COMMON_COPY } from '../common/copy';
import { useDraftCopy } from '../copy/useDraftCopy';
import { usePrivateMode } from '../privacy/PrivateModeProvider';
import { HIDDEN_AMOUNT, spokenMoney, walletMoneyText } from './walletMoney';
import type { MoneySign } from './walletMoney';

export type WalletAmountSize = 58 | 44 | 36 | 34 | 28 | 22 | 17 | 16 | 13;
export type WalletAmountTone = 'ink' | 'quiet' | 'violet' | 'under' | 'over';

export type WalletAmountProps = {
  /** Canonical satang. */
  minor: Minor;
  sign?: MoneySign;
  /** Keep `.00`. */
  exact?: boolean;
  size: WalletAmountSize;
  tone?: WalletAmountTone;
  /** Overrides the tone: the text colour of the material underneath. */
  color?: string;
  /** Said before the amount, such as ใช้ไปเดือนนี้. */
  label?: string;
  testID?: string;
};

/** The size of a canonical satang string. */
const magnitude = (minor: Minor): Minor => (minor.startsWith('-') ? minor.slice(1) : minor);

export function WalletAmount({ minor, sign = 'none', exact = false, size, tone, color, label, testID }: WalletAmountProps) {
  const colors = useWalletColors();
  const { hidden } = usePrivateMode();
  const t = useDraftCopy(COMMON_COPY);
  // The words follow what is drawn: only a balance says it is below zero; an
  // expense or income row says its kind in its own label.
  const spoken = hidden ? t('common.amountHidden') : spokenMoney(sign === 'balance' ? minor : magnitude(minor), t);
  // −0.02 em at 28 pt and above (the boards); the mask is spaced 0.12 em, as bw-private draws it.
  const letterSpacing = hidden ? (size * 12) / 100 : size >= 28 ? (size * -2) / 100 : undefined;
  return (
    <Text
      testID={testID}
      accessibilityLabel={label ? `${label} ${spoken}` : spoken}
      maxFontSizeMultiplier={amountMaxScale(size)}
      style={{
        fontFamily: LATIN_UI_FONT_FAMILY,
        fontWeight: '700',
        fontSize: size,
        lineHeight: walletLineHeight(size, 'latin'),
        letterSpacing,
        color: color ?? colors[tone ?? (sign === 'income' ? 'violet' : 'ink')],
      }}
    >
      {hidden ? HIDDEN_AMOUNT : walletMoneyText(minor, { sign, exact })}
    </Text>
  );
}
```

```tsx
// Private mode (W10, D41, R65): one device setting hides every amount behind
// ฿•••• until the eye is tapped again. It lives in the wallet's device store
// (P2-T3), so the sign-out erase forgets it; the tip stays seen once closed.
// Until the setting is read, amounts read as hidden and the eye waits, so an
// amount the person hid never shows for a moment at launch (as R66 treats an
// unknown motion answer). WalletRoot mounts the one provider; a catalogue
// section may mount its own over a store in memory.
import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState } from 'react';
import type { ReactNode } from 'react';
import { IntlContext } from 'react-intl';

import { trackPrivateModeToggled } from '@mobile/analytics/events';
import { isSupportedLocale } from '@mobile/i18n/locales';
import { announcePolitely } from '@mobile/ui/a11y';
import { tableT } from '../copy/draftText';
import { walletDeviceStore } from '../store/deviceWalletStore';
import type { KeyValueStore } from '../store/walletStore';
import { COPY } from './copy';

/** The device store's names: `jotbun.wallet.private-mode` and `jotbun.wallet.private-tip-seen`, `'1'` when on. */
export const PRIVATE_MODE_NAME = 'private-mode';
export const PRIVATE_TIP_NAME = 'private-tip-seen';

export type PrivateMode = {
  /** Every amount is drawn as ฿••••. True until the setting is read. */
  readonly hidden: boolean;
  /** The setting has been read. */
  readonly ready: boolean;
  /** The rule (R61): flips the setting, stores it, says it and sends one private_mode_toggled. Waits until `ready`. */
  readonly toggle: () => void;
  /** The tip was closed once, and never shows again. */
  readonly tipSeen: boolean;
  readonly dismissTip: () => void;
};

type Setting = Pick<PrivateMode, 'hidden' | 'ready' | 'tipSeen'>;

/** Outside the wallet root (a part drawn on its own): amounts show, and nothing changes. */
const OUTSIDE: PrivateMode = { hidden: false, ready: true, tipSeen: true, toggle: () => undefined, dismissTip: () => undefined };

/** Before the setting is read: hidden, with no tip yet. */
const UNREAD: Setting = { hidden: true, ready: false, tipSeen: true };

const PrivateModeContext = createContext<PrivateMode>(OUTSIDE);

/** A name the store cannot read counts as unset. */
const read = (store: KeyValueStore, name: string) => store.get(name).catch(() => null);

/** `store`: where this provider alone keeps the setting, held stable by its caller (a catalogue preview); the wallet's device store when absent. */
export function PrivateModeProvider({ children, store }: { children: ReactNode; store?: KeyValueStore }) {
  // What a tap says is in the app's language, from the root's IntlProvider.
  // WalletRoot's own test draws the root with none, so the context is read
  // without useIntl's throw, falling back to English as useActiveLocale does.
  const appLocale = (useContext(IntlContext) as { locale?: string } | null)?.locale;
  const t = useMemo(() => tableT(COPY, isSupportedLocale(appLocale) ? appLocale : 'en'), [appLocale]);
  const [setting, setSetting] = useState<Setting>(UNREAD);
  // What a tap flips: two taps before React renders still flip twice.
  const latest = useRef(setting);
  const publish = useCallback((next: Setting) => {
    latest.current = next;
    setSetting(next);
  }, []);
  const target = useCallback(() => store ?? walletDeviceStore(), [store]);

  useEffect(() => {
    // A run that has ended (unmounted, or React's development double run) drops its late read.
    let active = true;
    const from = target();
    void Promise.all([read(from, PRIVATE_MODE_NAME), read(from, PRIVATE_TIP_NAME)]).then(([mode, tip]) => {
      if (active) publish({ hidden: mode === '1', ready: true, tipSeen: tip === '1' });
    });
    return () => {
      active = false;
    };
  }, [target, publish]);

  const toggle = useCallback(() => {
    const now = latest.current;
    if (!now.ready) return;
    const hidden = !now.hidden;
    publish({ ...now, hidden });
    const to = target();
    // A choice that cannot be stored still holds until the app closes.
    void (hidden ? to.set(PRIVATE_MODE_NAME, '1') : to.remove(PRIVATE_MODE_NAME)).catch(() => undefined);
    announcePolitely(t(hidden ? 'privacy.hiddenAnnounce' : 'privacy.shownAnnounce'));
    trackPrivateModeToggled();
  }, [publish, target, t]);

  const dismissTip = useCallback(() => {
    publish({ ...latest.current, tipSeen: true });
    void target().set(PRIVATE_TIP_NAME, '1').catch(() => undefined);
  }, [publish, target]);

  const value = useMemo<PrivateMode>(() => ({ ...setting, toggle, dismissTip }), [setting, toggle, dismissTip]);
  return <PrivateModeContext.Provider value={value}>{children}</PrivateModeContext.Provider>;
}

/** Private mode as the nearest provider holds it; outside one, amounts show and nothing changes. */
export function usePrivateMode(): PrivateMode {
  return useContext(PrivateModeContext);
}
```

```tsx

// on the wallet's ground: money text (bw-amounts) and private mode
// (bw-private), from P2-T10; later kit tasks add theirs to this list.
//
// The catalogue's tests draw sections without the wallet root, and the app
// draws them inside it. So a section follows the private mode around it
// (outside a provider, amounts show), and a section that needs one state
// mounts its own PrivateModeProvider over a store kept in memory: the
// catalogue never changes the person's own setting. A section never imports
// the wallet test harness, which loads vitest.
import { useState } from 'react';
import type { ReactNode } from 'react';
import { View } from 'react-native';

import { useWalletColors } from '@mobile/theme/ThemeProvider';
import { BUN_RADIUS, BUN_SPACE } from '@mobile/ui/layout';
import { WalletAmount } from '@mobile/wallet/money/WalletAmount';
import type { WalletAmountSize } from '@mobile/wallet/money/WalletAmount';
import { PrivateEyeButton } from '@mobile/wallet/privacy/PrivateEyeButton';
import { PRIVATE_MODE_NAME, PrivateModeProvider } from '@mobile/wallet/privacy/PrivateModeProvider';
import { PrivateTip } from '@mobile/wallet/privacy/PrivateTip';
import type { KeyValueStore } from '@mobile/wallet/store/walletStore';
import { Specimen, SpecimenLabel, SpecimenRow } from '../parts';
import type { CatalogueSection } from '../registry';

const SIZES: readonly WalletAmountSize[] = [58, 44, 36, 34, 28, 22, 17, 16, 13];

/** A device store in memory, for a section that pins its own state. */
function memoryStore(seed: Readonly<Record<string, string>>): KeyValueStore {
  const kept = new Map(Object.entries(seed));
  return {
    get: async (name) => kept.get(name) ?? null,
    set: async (name, value) => {
      kept.set(name, value);
    },
    remove: async (name) => {
      kept.delete(name);
    },
    eraseAll: async () => {
      kept.clear();
    },
  };
}

function WalletGround({ children }: { children: ReactNode }) {
  const colors = useWalletColors();
  return (
    <View style={{ backgroundColor: colors.ground, borderRadius: BUN_RADIUS.card, padding: BUN_SPACE.md, gap: BUN_SPACE.sm }}>
      {children}
    </View>
  );
}

    </PrivateModeProvider>

}
```
