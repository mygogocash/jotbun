type CreateCommand = Extract<WalletCommand, { kind: 'create' }>;

/** A create whose outcome is not settled, or that says why it did not happen. */
type Sending =
  | { phase: 'idle' }
  | { phase: 'saving' }
  | { phase: 'unconfirmed'; command: CreateCommand; track: WalletTrack; message: string; blocked: boolean }
  | { phase: 'problem'; message: string; blocked: boolean };

export function NewWalletFlow({ returnTo, now }: { returnTo?: NewWalletReturn; now?: Date }) {
  const router = useRouter();
  const tc = useCopy();
  const t = useDraftCopy(LINES);
  const { accountId } = useDiaryView();
  const { setLook } = useWalletLooks();
  const commands = useWalletCommand();
  const today = todayInBangkok(now ?? new Date());
  // One wallet id per new wallet, fixed while the flow lives: a resend always names the same wallet.
  const [walletId] = useState(newId);
  const [step, setStep] = useState<1 | 2 | 3>(1);
  const [look, setChosenLook] = useState<WalletLook>({ kind: null, material: 'silk', face: 'smile' });
  const [form, setForm] = useState<WalletForm>(() => newWalletForm(today));
  const [keypad, setKeypad] = useState<KeypadState>(EMPTY_KEYPAD);
  const [errors, setErrors] = useState<WalletFieldErrors>({});
  const [sending, setSending] = useState<Sending>({ phase: 'idle' });
  const [added, setAdded] = useState<CreateWalletRequest | null>(null);
  // One command at a time, held by a ref: two presses before React draws again send one (P2-T19's review, m1).
  const held = useRef(false);

  // With no account open nothing could be kept: the wallet will have the look derived from its id.
  const shown = accountId === null ? derivedLook(walletId) : look;
  const typed = evaluateKeypad(keypad, { allowZero: true });
  const current: WalletForm = { ...form, openingMinor: typed.ok ? typed.minor : '' };
  const preview = walletPreview(current);
  const locked = sending.phase === 'saving' || sending.phase === 'unconfirmed';
  // P2-T35 adds /spend/setup; until it lands, typed routes do not know it, so the destination is typed by hand here.
  const destination = (returnTo ?? '/spend') as Href;

  const leave = () => {
    if (router.canGoBack()) router.back();
    else router.replace(destination);
  };
  const choose = (kind: WalletKind) => {
    setChosenLook({ kind, material: KIND_MATERIAL[kind], face: 'smile' });
    setStep(2);
  };
  const clear = (field: keyof WalletFieldErrors) => setErrors((previous) => {
    if (previous[field] === undefined) return previous;
    const next = { ...previous };
    delete next[field];
    return next;
  });

  async function send(command: CreateCommand, track: WalletTrack) {
    held.current = true;
    setErrors({});
    setSending({ phase: 'saving' });
    try {
      const outcome = await commands.send(command, track);
      if (outcome.kind === 'done') {
        // Kept only now the wallet is saved, as a fresh object (wallet-looks-unstored's review, n3).
        void setLook(command.input.id, { kind: look.kind, material: look.material, face: look.face });
        setSending({ phase: 'idle' });
        setAdded(command.input);
        setStep(3);
      } else if (outcome.kind === 'unconfirmed') {
        setSending({ phase: 'unconfirmed', command, track, message: outcome.message, blocked: outcome.blocked });
      } else if (outcome.kind === 'refused') {
        setErrors(outcome.fields);
        const named = Object.keys(outcome.fields).length > 0;
        setSending({ phase: 'problem', message: named ? tc('Check the highlighted fields.') : outcome.message ?? tc('This entry could not be saved.'), blocked: outcome.blocked });
      } else {
        // A create meets no other wallet's version; should Jotbun ever answer one, it was not saved.
        setSending({ phase: 'problem', message: tc('This entry could not be saved.'), blocked: false });
      }
    } finally {
      held.current = false;
    }
  }

  function add() {
    if (held.current) return;
    const built = buildCreateWallet(current, { id: walletId, today, tc });
    if (!built.ok) {
      setErrors(built.errors);
      return;
    }
    void send({ kind: 'create', slot: 'create', input: built.input }, { walletType: look.kind, material: shown.material, bunFace: shown.face });
  }

  // Try again: exactly the command whose outcome is unknown, so the same slot and payload go under the same key.
  function retry(pending: Extract<Sending, { phase: 'unconfirmed' }>) {
    if (held.current) return;
    void send(pending.command, pending.track);
  }

  if (step === 3 && added !== null) {
    return <Added input={added} look={shown} onDone={() => router.dismissTo(destination)} />;
  }
  if (step === 1) {
    return (
      <FlowPage back={{ label: t('common.back'), onPress: leave }} step={1}>
        <WalletText role="h1" accessibilityRole="header">{t('wallets.typeQuestion')}</WalletText>
        <WalletText role="body" tone="quietOnGround">{t('wallets.typeHint')}</WalletText>
        <KindList onKind={choose} />
      </FlowPage>
    );
  }
  const unconfirmed = sending.phase === 'unconfirmed' ? sending : null;
  const stuck = sending.phase === 'problem' && sending.blocked;
  const footer = (
    <View style={{ gap: 8 }}>
      {sending.phase === 'unconfirmed' || sending.phase === 'problem' ? (
        <View testID="new-wallet-problem" style={{ gap: 4, alignItems: 'flex-start' }}>
          <WalletText role="label" accessibilityRole="alert">{sending.message}</WalletText>
          {unconfirmed !== null && !unconfirmed.blocked ? (
            <TextLink label={t('common.tryAgain')} onPress={() => retry(unconfirmed)} />
          ) : null}
        </View>
      ) : null}
      <PrimaryButton testID="new-wallet-add" label={t('wallets.add')} busy={sending.phase === 'saving'} busyLabel={tc('Saving…')} disabled={locked || stuck} onPress={add} />
    </View>
  );
  return (
    <FlowPage back={{ label: t('common.back'), onPress: () => setStep(1), disabled: locked }} step={2} footer={footer}>
      <WalletText role="h1" accessibilityRole="header">{look.kind === null ? '' : t(KIND_WORDS[look.kind].name)}</WalletText>
      <WalletFields
        form={current}
        onForm={(next) => {
          setForm(next);
          clear('name');
        }}
        look={shown}
        onLook={setChosenLook}
        keypad={keypad}
        onKeypad={(next) => {
          setKeypad(next);
          clear('opening_minor');
        }}
        errors={errors}
        preview={{ id: walletId, name: preview.name ?? '', balanceMinor: preview.balanceMinor }}
        today={today}
        locked={locked}
      />
    </FlowPage>
  );
}

function EditForm({ wallet, hasRecords, today, notice, onNotice, onBack, onArchived, onChanged }: {
  wallet: WalletDto;
  hasRecords: HasRecords;
  today: string;
  notice: string | null;
  onNotice(notice: string | null): void;
  onBack(): void;
  onArchived(): void;
  onChanged(): void;
}) {
  const colors = useWalletColors();
  const insets = useSafeAreaInsets();
  const tc = useCopy();
  const t = useDraftCopy(LINES);
  const commands = useWalletCommand();
  const { lookOf, setLook } = useWalletLooks();
  const [form, setForm] = useState<WalletForm>(() => walletFormOf(wallet));
  // The opening balance as Jotbun holds it: masked in private mode until a key is pressed, and the first key starts it over (R106).
  const [keypad, setKeypad] = useState<KeypadState>(() => filledKeypad(wallet.opening_minor));
  const [errors, setErrors] = useState<WalletFieldErrors>({});
  const [sending, setSending] = useState<Sending>({ phase: 'idle' });
  const [asking, setAsking] = useState(false);
  const [archiveRefused, setArchiveRefused] = useState(false);
  // One command at a time, held by a ref: two presses before React draws again send one.
  const held = useRef(false);

  const typed = evaluateKeypad(keypad, { allowZero: true });
  const current: WalletForm = { ...form, openingMinor: typed.ok ? typed.minor : '' };
  const preview = walletPreview(current, wallet);
  const locked = sending.phase === 'saving' || sending.phase === 'unconfirmed';
  const stuck = sending.phase === 'problem' && sending.blocked;

  const clear = (field: keyof WalletFieldErrors) => setErrors((previous) => {
    if (previous[field] === undefined) return previous;
    const next = { ...previous };
    delete next[field];
    return next;
  });

  async function send(command: WalletCommand) {
    held.current = true;
    setSending({ phase: 'saving' });
    setArchiveRefused(false);
    onNotice(null);
    try {
      const outcome = await commands.send(command);
      if (outcome.kind === 'done') {
        setSending({ phase: 'idle' });
        if (command.kind === 'archive') onArchived();
        else onBack();
      } else if (outcome.kind === 'unconfirmed') {
        setSending({ phase: 'unconfirmed', command, message: outcome.message, blocked: outcome.blocked });
      } else if (outcome.kind === 'conflict') {
        // Changed on another device: say so, and read it again; a newer version starts the form over.
        setSending({ phase: 'idle' });
        onNotice(tc('This wallet changed on another device.'));
        onChanged();
      } else if (outcome.archiveRefused) {
        // The balance moved since it was read: say so, and read it again.
        setSending({ phase: 'idle' });
        setArchiveRefused(true);
        onChanged();
      } else if (outcome.needsReason) {
        // Jotbun wants the correction's reason: ask for it.
        setSending({ phase: 'idle' });
        setAsking(true);
      } else {
        setErrors(outcome.fields);
        const named = Object.keys(outcome.fields).length > 0;
        setSending({ phase: 'problem', message: named ? tc('Check the highlighted fields.') : outcome.message ?? tc('This entry could not be saved.'), blocked: outcome.blocked });
      }
    } finally {
      held.current = false;
    }
  }

  function save(reason: WalletCorrectionReason | null = form.reason) {
    if (held.current) return;
    const fields: WalletForm = { ...current, reason };
    // A correction to a wallet with entries (or not yet known to have none) asks why first; nothing is sent yet.
    if (reason === null && correctionNeedsReason(wallet, fields, hasRecords)) {
      setAsking(true);
      return;
    }
    const built = buildWalletUpdate(wallet, fields, { today, tc, hasRecords });
    if (!built.ok) {
      setErrors(built.errors);
      return;
    }
    if (built.unchanged) {
      onBack();
      return;
    }
    setErrors({});
    void send({ kind: 'update', slot: `edit:${built.id}`, id: built.id, version: built.version, changes: built.changes });
  }

  const answer = (reason: WalletCorrectionReason) => {
    setAsking(false);
    setForm((previous) => ({ ...previous, reason }));
    save(reason);
  };
  const archive = () => {
    if (held.current) return;
    void send({ kind: 'archive', slot: `archive:${wallet.id}`, id: wallet.id, version: wallet.version });
  };
  // Try again: exactly the command whose outcome is unknown, so it goes under the same key.
  const retry = (command: WalletCommand) => {
    if (held.current) return;
    void send(command);
  };

  return (
    <View testID="wallet-edit" style={{ flex: 1, backgroundColor: colors.ground }}>
      <ScrollView
        style={{ flex: 1 }}
        keyboardShouldPersistTaps="handled"
        contentContainerStyle={{ paddingTop: insets.top + 8, paddingHorizontal: 18, paddingBottom: insets.bottom + 24, gap: 14 }}
      >
        <EditHeader
          title={t('wallets.editTitle')}
          onBack={onBack}
          action={<SaveButton label={t('common.save')} busy={sending.phase === 'saving'} busyLabel={tc('Saving…')} disabled={locked || stuck} onPress={() => save()} />}
        />
        {notice === null ? null : <WalletText testID="wallet-edit-notice" role="label" accessibilityRole="alert">{notice}</WalletText>}
        {sending.phase === 'unconfirmed' || sending.phase === 'problem' ? (
          <View testID="wallet-edit-problem" style={{ gap: 4, alignItems: 'flex-start' }}>
            <WalletText role="label" accessibilityRole="alert">{sending.message}</WalletText>
            {sending.phase === 'unconfirmed' && !sending.blocked ? <TextLink label={t('common.tryAgain')} onPress={() => retry(sending.command)} /> : null}
          </View>
        ) : null}
        <WalletFields
          form={current}
          onForm={(next) => {
            setForm(next);
            clear('name');
          }}
          look={lookOf(wallet.id)}
          onLook={(next) => void setLook(wallet.id, { kind: next.kind, material: next.material, face: next.face })}
          keypad={keypad}
          onKeypad={(next) => {
            setKeypad(next);
            clear('opening_minor');
          }}
          errors={errors}
          preview={{ id: wallet.id, name: preview.name ?? '', balanceMinor: preview.balanceMinor }}
          today={today}
          locked={locked}
        />
        {hasRecords === true ? <WalletText testID="wallet-has-records" role="caption" tone="quietOnGround">{t('wallets.hasRecords')}</WalletText> : null}
        <Archiving wallet={wallet} refused={archiveRefused} disabled={locked} onArchive={archive} />
      </ScrollView>
      <CorrectionTray visible={asking} onClose={() => setAsking(false)} onReason={answer} />
    </View>
  );
