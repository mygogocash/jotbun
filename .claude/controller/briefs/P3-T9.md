```ts
export async function askWithEscalation(input, deps) {
  // H3: every Haiku turn shares one child deadline, so Opus keeps the rest of the route's.
  const haiku = await runAskConversation(HAIKU, input, { ...deps, deadline: deps.deadline.child(HAIKU_ASK_CAP_MS) })
  if (haiku.kind === 'grounded') return { answer: haiku.answer, figures: haiku.figures, source: 'ai', model: HAIKU, escalated: false }
  if (haiku.kind === 'failed') return plainFigures(haiku, [], input.locale, false)
  const opus = await runAskConversation(OPUS, input, deps)
  if (opus.kind === 'grounded') return { answer: opus.answer, figures: opus.figures, source: 'ai', model: OPUS, escalated: true }
  return plainFigures(opus, haiku.figures, input.locale, true)
}

function plainFigures(last, earlier, locale, escalated) {
  const figures = last.figures.length > 0 ? last.figures : earlier
  if (last.kind === 'failed' && figures.length === 0) return { paused: true }
  return { answer: figuresText(figures, locale), figures, source: 'facts', model: 'rules', escalated }
}
```

`runAskConversation` is the route's earlier loop, moved: its own messages (starting from the person's question alone, the same JSON as before), its own `AllowedNumbers` and figures, `opusRequest('medium', 12_000)` or `haikuRequest('medium', 12_000)` by tier, `escalated: tier === OPUS` on each call. A failed call is `failed` (logged by class name), and a `DeadlinePassed` is `no_answer`; a refusal is `declined`; a cut-short or empty answer is `no_answer`; the turn limit is `no_answer`; a grounded text `grounded`, any other `ungrounded`.

The Ask Bun route in `apps/edge/src/v1/ai/service.ts`:

```ts
  async ask(s: VaultSession, request: AiAskRequest): Promise<AiAskResult> {
    const { trace, deadline } = this.start('ask')
    const ctx = await step<AiContext>(s, 'ai.context')
    if (!ctx.consent) throw new ProblemException(403, 'AI_CONSENT_REQUIRED')
    if (!this.deps.transport) throw this.paused()
    const use = await this.consume(s, 'ask')
    const today = trustedToday(request.today, bangkokDate(this.now()))
    const categories = ctx.categories.map((category) => ({ id: category.id, name: category.label ?? category.builtin_key }))
    const answered = await askWithEscalation(
      { question: request.question, today, locale: request.locale, categories },
      {
        transport: this.deps.transport,
        trace,
        deadline,
        tool: (name, input) => step<AiToolReply>(s, 'ai.tool', { body: { name, input, locale: request.locale } }),
      },
    )
    // H5 (P3-T7's carry, R116): the use goes back when no model's words are shown (the plain figures or 503) and
    // the last call failed with an outage. A grounded answer's last call answered, so it never gives the use back.
    await this.giveBack(s, use, trace)
    if ('paused' in answered) throw this.paused()
    return answered
  }
```

R116 in `apps/edge/src/v1/ai/uses.ts`:

```ts
export function shouldRefund(calls: readonly RecordedCall[], answeredByModel: boolean): boolean {
  const last = calls.at(-1)
  if (answeredByModel || last === undefined || last.failure === null || !GIVES_BACK.includes(last.failure)) return false
  return last.sent || !calls.some((call) => call.failure === null)
}
```
