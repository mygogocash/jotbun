## Interfaces (as `real/` has them)

```ts
// src/v1/ai/transport.ts
export const AI_GATEWAY_HEADERS = {
  'cf-aig-collect-log': 'false',
  'cf-aig-skip-cache': 'true',
} as const

export const MIN_CALL_MS = 2000

export type ModelTransport = {
  send(options: {
    messages: Anthropic.Beta.Messages.BetaMessageParam[]
    tools?: Anthropic.Beta.Messages.BetaTool[]
    system?: string
    maxTokens?: number
    timeoutMs: number
  }): Promise<Anthropic.Beta.Messages.BetaMessage>
}

export function anthropicTransport(options: {
  apiKey: string
  baseURL?: string
  gatewayToken?: string
  fetch?: typeof fetch
}): ModelTransport

export function transportFromEnv(env: V1Env): ModelTransport | null

export type CallFailure =
  | { kind: 'rate_limited'; waitSeconds?: number }
  | { kind: 'timeout'; elapsedMs: number }
  | { kind: 'rejected'; status: number; message: string }
  | { kind: 'network'; message: string }

export function callFailure(error: unknown, elapsedMs: number): CallFailure

export class DeadlinePassed extends Error {
  readonly code = 'AI_DEADLINE_PASSED'
  constructor(message = 'Request AI deadline passed') {
    super(message)
    this.name = 'DeadlinePassed'
  }
}

export class AiDeadline {
  readonly startMs: number
  readonly limitMs: number
  constructor(limitMs: number, startMs?: number)
  get remainingMs(): number
  get passed(): boolean
  child(capMs: number): AiDeadline
  callTimeoutMs(minRemainingMs?: number): number
}

export type RecordedCall = {
  callIndex: number
  model: string
  requestTokens: number
  responseTokens: number
  elapsedMs: number
  cached: boolean
  failure?: CallFailure
}

export type AiTrace = {
  readonly calls: readonly RecordedCall[]
  record(call: Omit<RecordedCall, 'callIndex'>): RecordedCall
  summary(): { totalCalls: number; totalElapsedMs: number; totalTokens: number }
}

export function newTrace(): AiTrace

export async function tracedSend(options: {
  transport: ModelTransport
  trace: AiTrace
  deadline: AiDeadline
  callCapMs?: number
  model?: string
  messages: Anthropic.Beta.Messages.BetaMessageParam[]
  tools?: Anthropic.Beta.Messages.BetaTool[]
  system?: string
  maxTokens?: number
}): Promise<Anthropic.Beta.Messages.BetaMessage>
```

## Planner rulings (the calls this brief makes, with the reason for each)

- **P1. Anthropic SDK 1.2.0 auth header.** `@anthropic-ai/sdk` v1.2.0 sends API keys via `Authorization: Bearer <apiKey>` header instead of legacy `x-api-key`. As specified in Interfaces, `defaultHeaders` contains `AI_GATEWAY_HEADERS` plus optional `cf-aig-authorization: Bearer <gatewayToken>`. The unit test accepts either header (`Authorization: Bearer <apiKey>` or `x-api-key`).
- **P2. `WorkerContext.transport` optionality.** `WorkerContext.transport` is declared optional (`transport?: ModelTransport | null`) to maintain backward compatibility with non-AI handlers (such as `confirmRoute` in `imports-review.test.ts`) that construct mock contexts without transport. In `runWorker`, it is always supplied as `deps.transport !== undefined ? deps.transport : transportFromEnv(env)`. In `ai/routes.ts`, `aiOf` passes `w.transport ?? null` to `VaultAiService`.
- **P3. Removal of legacy `SendMessage` type and `anthropicSender`.** The old direct-fetch sender is replaced by `ModelTransport`. `AiModelClient` wraps Anthropic SDK with transport fetch. Base request builders and serialization utilities are preserved.
- **P4. Merge proof disjointness with PRs #406 and #407.** PR #406 (P2-T33) and PR #407 (P2-T14) modify schema and app code outside `apps/edge`. A strict git merge-base diff confirmed 0 overlapping files (`PL/merge/merge-proof.md`).
